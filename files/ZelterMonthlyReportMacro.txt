Option Explicit

' ================================================================
'  Z E L T E R   M O N T H L Y   R E P O R T   M A C R O
' ================================================================
'
'  Run this macro from the sheet that contains the raw data.
'
'  TAB OUTPUTS:
'  - Spending Data  : cleaned full transaction list
'  - Discarded      : every row removed (internal transfers,
'                     qualifying activity types, APO dividends)
'  - Deposits       : positive-amount rows only, by account
'  - Withdrawals    : negative-amount rows only, by account
'
' ================================================================

' ----------------------------------------------------------------
'  Returns a combined array of all activity types to move to
'  Discarded.  Split across three sub-arrays to stay under VBA's
'  24 line-continuation limit.
'  "Qualified Dividend" intentionally excluded - handled in Step 6.
' ----------------------------------------------------------------
Private Function DiscardActivityList() As Variant
    Dim a1 As Variant
    Dim a2 As Variant
    Dim a3 As Variant

    a1 = Array( _
        "Bought", _
        "Dividend", _
        "Dividend Reinvestment", _
        "Interest Income", _
        "Service Fee", _
        "Service Fee Adj", _
        "Sold", _
        "Stock Split", _
        "Tax Exempt Dividend", _
        "Tax Exempt Interest Income", _
        "Account Fee", _
        "Account Fee Adj", _
        "Automated Payment-Adj")

    a2 = Array( _
        "Cash in Lieu", _
        "LT Cap Gain Distribution", _
        "Return of Capital", _
        "Security Sold", _
        "Sold - Adjusted", _
        "Sold - Canceled", _
        "Stock Spin-Off", _
        "Return of Principal", _
        "Interest Income-Adj", _
        "Cancel Sell", _
        "Dividend - Adjustment", _
        "Miscellaneous Income", _
        "Margin Int Charged-Adj")

    a3 = Array( _
        "Margin Interest Charged", _
        "Policy Networked", _
        "Redemption", _
        "Transfer into Account", _
        "Transfer out of Account", _
        "Cashless Dividend", _
        "Exchange Deliver Out", _
        "Exchange Received In", _
        "Tax Exempt Income-Adj", _
        "Tax Withholding", _
        "Tax Withholding Adj", _
        "Tax Exempt Int Income-Adj")

    ' Merge all three into one flat array
    Dim total As Long
    total = UBound(a1) + UBound(a2) + UBound(a3) + 3  ' +3 for the three LBound=0 arrays

    Dim combined() As String
    ReDim combined(0 To total - 1)

    Dim idx As Long
    idx = 0
    Dim i As Long

    For i = 0 To UBound(a1)
        combined(idx) = CStr(a1(i))
        idx = idx + 1
    Next i
    For i = 0 To UBound(a2)
        combined(idx) = CStr(a2(i))
        idx = idx + 1
    Next i
    For i = 0 To UBound(a3)
        combined(idx) = CStr(a3(i))
        idx = idx + 1
    Next i

    DiscardActivityList = combined
End Function


' ================================================================
'  MAIN MACRO
' ================================================================
Sub ZelterMonthlyReportMacro()

    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    On Error GoTo ErrorHandler

    Dim wsSource As Worksheet
    Set wsSource = ActiveSheet

    ' ============================================================
    ' STEP 1 - Find "Activity Date" in column A; delete rows above
    ' ============================================================
    Dim hdrRow As Long
    hdrRow = 0
    Dim i As Long
    For i = 1 To 10000
        If Trim(CStr(wsSource.Cells(i, 1).Value)) = "Activity Date" Then
            hdrRow = i
            Exit For
        End If
    Next i

    If hdrRow = 0 Then
        MsgBox "ERROR: 'Activity Date' not found in column A." & vbCrLf & _
               "Macro aborted.", vbCritical
        GoTo Cleanup
    End If

    If hdrRow > 1 Then
        wsSource.Rows("1:" & hdrRow - 1).Delete
    End If

    ' ============================================================
    ' STEP 2 - Find first empty row in col A; delete it + below
    ' ============================================================
    Dim dLast As Long
    dLast = 1
    Do While Trim(CStr(wsSource.Cells(dLast + 1, 1).Value)) <> ""
        dLast = dLast + 1
        If dLast > 1000000 Then Exit Do
    Loop

    Dim emptyStart As Long
    emptyStart = dLast + 1
    If emptyStart <= wsSource.Rows.Count Then
        wsSource.Rows(emptyStart & ":" & wsSource.Rows.Count).Delete
    End If

    ' ============================================================
    ' STEP 3 - Duplicate raw sheet as "Spending Data"
    '          Pre-delete any stale output tabs
    ' ============================================================
    Application.DisplayAlerts = False

    Dim shNames(1 To 4) As String
    shNames(1) = "Spending Data"
    shNames(2) = "Discarded"
    shNames(3) = "Deposits"
    shNames(4) = "Withdrawals"

    Dim ws As Worksheet
    Dim si As Long
    For si = 1 To 4
        For Each ws In ThisWorkbook.Sheets
            If ws.Name = shNames(si) Then
                ws.Delete
                Exit For
            End If
        Next ws
    Next si

    Application.DisplayAlerts = True

    wsSource.Copy After:=wsSource
    Dim wsSD As Worksheet
    Set wsSD = ActiveSheet
    wsSD.Name = "Spending Data"

    ' ============================================================
    ' Resolve column indices from header row 1
    ' ============================================================
    Dim nCols As Long
    nCols = wsSD.Cells(1, wsSD.Columns.Count).End(xlToLeft).Column

    Dim cActDate   As Long
    Dim cTransDate As Long
    Dim cAcct      As Long
    Dim cDesc      As Long
    Dim cActivity  As Long
    Dim cCheckNum  As Long
    Dim cCardNum   As Long
    Dim cSymbol    As Long
    Dim cCusip     As Long
    Dim cTags      As Long
    Dim cQty       As Long
    Dim cPrice     As Long
    Dim cAmt       As Long

    Dim col As Long
    For col = 1 To nCols
        Select Case Trim(CStr(wsSD.Cells(1, col).Value))
            Case "Activity Date":    cActDate   = col
            Case "Transaction Date": cTransDate = col
            Case "Account":          cAcct      = col
            Case "Description":      cDesc      = col
            Case "Activity":         cActivity  = col
            Case "Check Number":     cCheckNum  = col
            Case "Card Number":      cCardNum   = col
            Case "Symbol":           cSymbol    = col
            Case "Cusip":            cCusip     = col
            Case "Tags":             cTags      = col
            Case "Quantity":         cQty       = col
            Case "Price":            cPrice     = col
            Case "Amount($)":        cAmt       = col
        End Select
    Next col

    If cActivity = 0 Or cAcct = 0 Or cAmt = 0 Then
        MsgBox "ERROR: Required columns (Activity, Account, Amount($)) not found." & vbCrLf & _
               "Verify header names match exactly.", vbCritical
        GoTo Cleanup
    End If

    ' ============================================================
    ' STEP 4 - Apply AutoFilter on header row
    ' ============================================================
    wsSD.Rows(1).AutoFilter

    ' ============================================================
    ' STEP 5 - Create Discarded tab; move all qualifying activities
    '          "Qualified Dividend" is intentionally skipped here
    ' ============================================================
    Dim wsDiscard As Worksheet
    Set wsDiscard = ThisWorkbook.Sheets.Add( _
        After:=ThisWorkbook.Sheets(ThisWorkbook.Sheets.Count))
    wsDiscard.Name = "Discarded"
    wsSD.Rows(1).Copy wsDiscard.Rows(1)

    Dim discardNext As Long
    discardNext = 2

    Dim dList As Variant
    dList = DiscardActivityList()

    Dim rLast As Long
    rLast = wsSD.Cells(wsSD.Rows.Count, 1).End(xlUp).Row

    Dim r As Long
    Dim actStr As String
    Dim d As Long
    For r = rLast To 2 Step -1
        actStr = Trim(CStr(wsSD.Cells(r, cActivity).Value))
        If LCase(actStr) = "qualified dividend" Then GoTo Skip5
        For d = 0 To UBound(dList)
            If LCase(actStr) = LCase(CStr(dList(d))) Then
                wsSD.Rows(r).Copy wsDiscard.Rows(discardNext)
                discardNext = discardNext + 1
                wsSD.Rows(r).Delete
                Exit For
            End If
        Next d
        Skip5:
    Next r

    ' ============================================================
    ' STEP 6 - Filter to Qualified Dividend; remove Symbol = "APO"
    ' ============================================================
    wsSD.AutoFilterMode = False
    rLast = wsSD.Cells(wsSD.Rows.Count, 1).End(xlUp).Row

    wsSD.Range(wsSD.Cells(1, 1), wsSD.Cells(rLast, nCols)).AutoFilter _
        Field:=cActivity, Criteria1:="Qualified Dividend"

    For r = rLast To 2 Step -1
        If Not wsSD.Rows(r).Hidden Then
            If cSymbol > 0 Then
                If Trim(CStr(wsSD.Cells(r, cSymbol).Value)) = "APO" Then
                    wsSD.Rows(r).Copy wsDiscard.Rows(discardNext)
                    discardNext = discardNext + 1
                    wsSD.Rows(r).Delete
                End If
            End If
        End If
    Next r

    ' ============================================================
    ' STEP 7 - Show all / reset filter
    ' ============================================================
    wsSD.AutoFilterMode = False
    rLast = wsSD.Cells(wsSD.Rows.Count, 1).End(xlUp).Row
    wsSD.Range(wsSD.Cells(1, 1), wsSD.Cells(1, nCols)).AutoFilter

    ' ============================================================
    ' STEP 8 - Cash Transfer matching
    '
    '   For each Cash Transfer row P, find an unmatched row Q where:
    '     - Last 4 chars of Account(P) appear in the last line of
    '       Description(Q)
    '     - Last 4 chars of Account(Q) appear in the last line of
    '       Description(P)
    '     - Amount(P) + Amount(Q) nets to 0
    '   Matched pairs -> Discarded.  Unmatched -> stay.
    '   Multiple transfers between the same accounts are paired
    '   greedily (first available match wins).
    ' ============================================================
    wsSD.AutoFilterMode = False
    rLast = wsSD.Cells(wsSD.Rows.Count, 1).End(xlUp).Row

    wsSD.Range(wsSD.Cells(1, 1), wsSD.Cells(rLast, nCols)).AutoFilter _
        Field:=cActivity, Criteria1:="Cash Transfer"

    Dim ctRows() As Long
    Dim ctN As Long
    ctN = 0
    ReDim ctRows(1 To rLast)
    For r = 2 To rLast
        If Not wsSD.Rows(r).Hidden Then
            ctN = ctN + 1
            ctRows(ctN) = r
        End If
    Next r
    wsSD.AutoFilterMode = False

    If ctN >= 2 Then
        Dim ctMatched() As Boolean
        ReDim ctMatched(1 To ctN)

        Dim p As Long
        Dim q As Long
        Dim acctP As String
        Dim acctQ As String
        Dim last4P As String
        Dim last4Q As String
        Dim amtP As Double
        Dim amtQ As Double
        Dim descP As String
        Dim descQ As String
        Dim lastLineP As String
        Dim lastLineQ As String
        Dim linesP() As String
        Dim linesQ() As String
        Dim li As Long

        For p = 1 To ctN
            If Not ctMatched(p) Then

                acctP  = Trim(CStr(wsSD.Cells(ctRows(p), cAcct).Value))
                last4P = Right(acctP, 4)
                If Not IsNumeric(wsSD.Cells(ctRows(p), cAmt).Value) Then GoTo NextCT_P
                amtP = CDbl(wsSD.Cells(ctRows(p), cAmt).Value)

                descP = CStr(wsSD.Cells(ctRows(p), cDesc).Value)
                descP = Replace(descP, Chr(13) & Chr(10), Chr(10))
                descP = Replace(descP, Chr(13), Chr(10))
                linesP = Split(descP, Chr(10))
                lastLineP = ""
                For li = UBound(linesP) To 0 Step -1
                    If Trim(linesP(li)) <> "" Then
                        lastLineP = Trim(linesP(li))
                        Exit For
                    End If
                Next li

                For q = p + 1 To ctN
                    If Not ctMatched(q) Then

                        acctQ  = Trim(CStr(wsSD.Cells(ctRows(q), cAcct).Value))
                        last4Q = Right(acctQ, 4)
                        If Not IsNumeric(wsSD.Cells(ctRows(q), cAmt).Value) Then GoTo NextCT_Q
                        amtQ = CDbl(wsSD.Cells(ctRows(q), cAmt).Value)

                        descQ = CStr(wsSD.Cells(ctRows(q), cDesc).Value)
                        descQ = Replace(descQ, Chr(13) & Chr(10), Chr(10))
                        descQ = Replace(descQ, Chr(13), Chr(10))
                        linesQ = Split(descQ, Chr(10))
                        lastLineQ = ""
                        For li = UBound(linesQ) To 0 Step -1
                            If Trim(linesQ(li)) <> "" Then
                                lastLineQ = Trim(linesQ(li))
                                Exit For
                            End If
                        Next li

                        If Len(last4P) > 0 And Len(last4Q) > 0 Then
                            If InStr(lastLineQ, last4P) > 0 And _
                               InStr(lastLineP, last4Q) > 0 And _
                               Abs(amtP + amtQ) < 0.01 Then
                                ctMatched(p) = True
                                ctMatched(q) = True
                                Exit For
                            End If
                        End If
                        NextCT_Q:
                    End If
                Next q
                NextCT_P:
            End If
        Next p

        Dim toDelete() As Long
        Dim delN As Long
        delN = 0
        ReDim toDelete(1 To ctN)
        For p = 1 To ctN
            If ctMatched(p) Then
                delN = delN + 1
                toDelete(delN) = ctRows(p)
            End If
        Next p

        Dim tmpRow As Long
        For p = 1 To delN - 1
            For q = p + 1 To delN
                If toDelete(p) < toDelete(q) Then
                    tmpRow      = toDelete(p)
                    toDelete(p) = toDelete(q)
                    toDelete(q) = tmpRow
                End If
            Next q
        Next p

        For p = 1 To delN
            wsSD.Rows(toDelete(p)).Copy wsDiscard.Rows(discardNext)
            discardNext = discardNext + 1
            wsSD.Rows(toDelete(p)).Delete
        Next p
    End If

    ' ============================================================
    ' STEP 9 - Remove all filters
    ' ============================================================
    wsSD.AutoFilterMode = False

    ' ============================================================
    ' STEP 10 - Sort all data A-Z by Account
    ' ============================================================
    rLast = wsSD.Cells(wsSD.Rows.Count, 1).End(xlUp).Row
    wsSD.Range(wsSD.Cells(1, 1), wsSD.Cells(1, nCols)).AutoFilter

    With wsSD.Sort
        .SortFields.Clear
        .SortFields.Add Key:=wsSD.Columns(cAcct), _
                        SortOn:=xlSortOnValues, Order:=xlAscending
        .SetRange wsSD.Range(wsSD.Cells(1, 1), wsSD.Cells(rLast, nCols))
        .Header = xlYes
        .Apply
    End With

    ' ============================================================
    ' STEP 11 - Insert 3 separator rows between account groups;
    '           place a header copy directly above each new group
    '
    '   Layout between Group A and Group B:
    '     [last row of Group A]
    '     [blank spacer]
    '     [blank spacer]
    '     [header copy]      <- sits directly above Group B
    '     [first row of Group B]
    ' ============================================================
    wsSD.AutoFilterMode = False
    rLast = wsSD.Cells(wsSD.Rows.Count, 1).End(xlUp).Row

    Dim thisAcct As String
    Dim prevAcct As String
    For r = rLast To 3 Step -1
        thisAcct = Trim(CStr(wsSD.Cells(r, cAcct).Value))
        prevAcct = Trim(CStr(wsSD.Cells(r - 1, cAcct).Value))

        If thisAcct <> "" And prevAcct <> "" And thisAcct <> prevAcct Then
            wsSD.Rows(r).Resize(3).Insert Shift:=xlDown
            wsSD.Rows(1).Copy wsSD.Rows(r + 2)
        End If
    Next r

    ' ============================================================
    ' STEP 12 - Create Deposits and Withdrawals tabs
    ' ============================================================
    wsSD.Copy After:=wsSD
    Dim wsDeposits As Worksheet
    Set wsDeposits = ActiveSheet
    wsDeposits.Name = "Deposits"

    wsSD.Copy After:=wsDeposits
    Dim wsWithdrawals As Worksheet
    Set wsWithdrawals = ActiveSheet
    wsWithdrawals.Name = "Withdrawals"

    Call CleanAmountTab(wsDeposits, cAmt, True)
    Call CleanAmountTab(wsWithdrawals, cAmt, False)

    Dim hasAPO As Boolean
    hasAPO = False
    Dim depEnd As Long
    depEnd = wsDeposits.Cells(wsDeposits.Rows.Count, 1).End(xlUp).Row
    If cSymbol > 0 Then
        For r = 2 To depEnd
            If Trim(CStr(wsDeposits.Cells(r, cSymbol).Value)) = "APO" Then
                hasAPO = True
                Exit For
            End If
        Next r
    End If

    Call DeleteSpecifiedColumns(wsDeposits, _
                                cTransDate, cCheckNum, cCardNum, _
                                cSymbol, cCusip, cTags, cQty, cPrice, _
                                Not hasAPO)

    Call DeleteSpecifiedColumns(wsWithdrawals, _
                                cTransDate, cCheckNum, cCardNum, _
                                0, 0, cTags, 0, cPrice, _
                                False)

    ' ============================================================
    ' STEP 13 - Summary tables
    ' ============================================================
    Call AddSummaryTable(wsDeposits)
    Call AddSummaryTable(wsWithdrawals)

    Application.ScreenUpdating = True
    Application.Calculation = xlCalculationAutomatic
    MsgBox "ZelterMonthlyReportMacro completed successfully!", vbInformation
    Exit Sub

ErrorHandler:
    Application.ScreenUpdating = True
    Application.Calculation = xlCalculationAutomatic
    MsgBox "Error " & Err.Number & ": " & Err.Description, vbCritical
    Exit Sub

Cleanup:
    Application.ScreenUpdating = True
    Application.Calculation = xlCalculationAutomatic

End Sub


' ================================================================
'  HELPER : CleanAmountTab
' ================================================================
Private Sub CleanAmountTab(ws As Worksheet, cAmt As Long, keepPositive As Boolean)

    Dim r As Long
    Dim rLast As Long
    rLast = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row

    Dim cellA As String
    Dim amtV As Double
    Dim doDelete As Boolean

    For r = rLast To 2 Step -1
        cellA = Trim(CStr(ws.Cells(r, 1).Value))
        If cellA = "" Or cellA = "Activity Date" Then GoTo CleanNext1
        If Not IsNumeric(ws.Cells(r, cAmt).Value) Then GoTo CleanNext1

        amtV = CDbl(ws.Cells(r, cAmt).Value)
        doDelete = False
        If keepPositive     And amtV < 0 Then doDelete = True
        If Not keepPositive And amtV > 0 Then doDelete = True
        If doDelete Then ws.Rows(r).Delete
        CleanNext1:
    Next r

    rLast = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    Dim startBlock As Long
    Dim hasData As Boolean
    Dim fwd As Long
    Dim fwdVal As String

    For r = rLast To 2 Step -1
        If r > 1 And Trim(CStr(ws.Cells(r, 1).Value)) = "Activity Date" Then
            hasData = False
            For fwd = r + 1 To rLast
                fwdVal = Trim(CStr(ws.Cells(fwd, 1).Value))
                If fwdVal = "Activity Date" Then Exit For
                If fwdVal <> "" Then
                    hasData = True
                    Exit For
                End If
            Next fwd

            If Not hasData Then
                startBlock = r
                Do While startBlock > 2
                    If Trim(CStr(ws.Cells(startBlock - 1, 1).Value)) = "" Then
                        startBlock = startBlock - 1
                    Else
                        Exit Do
                    End If
                Loop
                ws.Rows(startBlock & ":" & r).Delete
                rLast = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
            End If
        End If
    Next r

    rLast = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    Do While rLast >= 2
        If Trim(CStr(ws.Cells(2, 1).Value)) = "" Then
            ws.Rows(2).Delete
            rLast = rLast - 1
        Else
            Exit Do
        End If
    Loop

End Sub


' ================================================================
'  HELPER : DeleteSpecifiedColumns
' ================================================================
Private Sub DeleteSpecifiedColumns(ws As Worksheet, _
    cTransDate As Long, cCheckNum As Long, cCardNum As Long, _
    cSymbol As Long, cCusip As Long, cTags As Long, _
    cQty As Long, cPrice As Long, deleteSymbolGroup As Boolean)

    Dim colList(1 To 8) As Long
    Dim n As Long
    n = 0

    If cTransDate > 0 Then: n = n + 1: colList(n) = cTransDate
    If cCheckNum  > 0 Then: n = n + 1: colList(n) = cCheckNum
    If cCardNum   > 0 Then: n = n + 1: colList(n) = cCardNum
    If cTags      > 0 Then: n = n + 1: colList(n) = cTags
    If cPrice     > 0 Then: n = n + 1: colList(n) = cPrice
    If deleteSymbolGroup Then
        If cSymbol > 0 Then: n = n + 1: colList(n) = cSymbol
        If cCusip  > 0 Then: n = n + 1: colList(n) = cCusip
        If cQty    > 0 Then: n = n + 1: colList(n) = cQty
    End If

    If n = 0 Then Exit Sub

    Dim a As Long
    Dim b As Long
    Dim tmp As Long
    For a = 1 To n - 1
        For b = a + 1 To n
            If colList(a) < colList(b) Then
                tmp        = colList(a)
                colList(a) = colList(b)
                colList(b) = tmp
            End If
        Next b
    Next a

    For a = 1 To n
        If colList(a) > 0 Then ws.Columns(colList(a)).Delete
    Next a

End Sub


' ================================================================
'  HELPER : AddSummaryTable
' ================================================================
Private Sub AddSummaryTable(ws As Worksheet)

    Dim rLast As Long
    Dim cLast As Long
    rLast = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    cLast = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column

    If rLast < 2 Or cLast < 2 Then Exit Sub

    Dim colAcct As Long
    Dim colAmt  As Long
    Dim c As Long
    For c = 1 To cLast
        Select Case Trim(CStr(ws.Cells(1, c).Value))
            Case "Account":   colAcct = c
            Case "Amount($)": colAmt  = c
        End Select
    Next c
    If colAcct = 0 Or colAmt = 0 Then Exit Sub

    Dim accts(1 To 500) As String
    Dim amts(1 To 500)  As Double
    Dim n As Long
    n = 0

    Dim r As Long
    Dim v1 As String
    Dim acctName As String
    Dim amtVal   As Double
    Dim found    As Boolean
    Dim k As Long

    For r = 2 To rLast
        v1 = Trim(CStr(ws.Cells(r, 1).Value))
        If v1 = "" Or v1 = "Activity Date" Then GoTo SumNext
        If Not IsNumeric(ws.Cells(r, colAmt).Value) Then GoTo SumNext

        acctName = Trim(CStr(ws.Cells(r, colAcct).Value))
        If acctName = "" Or acctName = "Account" Then GoTo SumNext

        amtVal = CDbl(ws.Cells(r, colAmt).Value)
        found = False

        For k = 1 To n
            If accts(k) = acctName Then
                amts(k) = amts(k) + amtVal
                found = True
                Exit For
            End If
        Next k

        If Not found Then
            n = n + 1
            accts(n) = acctName
            amts(n)  = amtVal
        End If
        SumNext:
    Next r

    If n = 0 Then Exit Sub

    Dim startRow As Long
    Dim leftCol  As Long
    Dim rightCol As Long
    startRow = rLast + 3
    leftCol  = cLast - 1
    rightCol = cLast

    ws.Cells(startRow, leftCol).Value  = "Account"
    ws.Cells(startRow, rightCol).Value = "Amount($)"
    ws.Cells(startRow, leftCol).Font.Bold  = True
    ws.Cells(startRow, rightCol).Font.Bold = True

    For k = 1 To n
        ws.Cells(startRow + k, leftCol).Value  = accts(k)
        ws.Cells(startRow + k, rightCol).Value = amts(k)
        ws.Cells(startRow + k, rightCol).NumberFormat = "#,##0.00"
    Next k

    Dim sumRng As Range
    Set sumRng = ws.Range( _
        ws.Cells(startRow, leftCol), _
        ws.Cells(startRow + n, rightCol))

    With sumRng.Borders
        .LineStyle = xlContinuous
        .Weight    = xlThin
        .Color     = RGB(0, 0, 0)
    End With

    ws.Columns(leftCol).AutoFit
    ws.Columns(rightCol).AutoFit

End Sub
