Option Explicit

' ================================================================
'  Z E L T E R   M O N T H L Y   R E P O R T   M A C R O
' ================================================================
'
'  Run from the sheet containing the raw downloaded data.
'
'  OUTPUT TABS:
'  - Spending Data  : full cleaned & sorted transaction list
'  - Discarded      : every row removed and why
'  - Deposits       : positive amounts only, grouped by account
'  - Withdrawals    : negative amounts only, grouped by account
'
' ================================================================

' ----------------------------------------------------------------
'  39 activity types that get moved to Discarded.
'  Split across three Array() calls to stay under VBA's hard
'  limit of 24 line-continuation characters per statement.
'  "Qualified Dividend" is intentionally absent - handled
'  separately in Step 6 (only discarded when Symbol = "APO").
' ----------------------------------------------------------------
Private Function DiscardActivityList() As Variant

    Dim a1 As Variant, a2 As Variant, a3 As Variant

    a1 = Array( _
        "Bought", _
        "Dividend", _
        "Dividend Reinvestment", _
        "Interest Income", _
        "Service Fee", _
        "Service Fee Adj", _
        "Sold", _
        "Stock Split", _
        "Tax Exempt Dividend", _
        "Tax Exempt Interest Income", _
        "Account Fee", _
        "Account Fee Adj", _
        "Automated Payment-Adj")

    a2 = Array( _
        "Cash in Lieu", _
        "LT Cap Gain Distribution", _
        "Return of Capital", _
        "Security Sold", _
        "Sold - Adjusted", _
        "Sold - Canceled", _
        "Stock Spin-Off", _
        "Return of Principal", _
        "Interest Income-Adj", _
        "Cancel Sell", _
        "Dividend - Adjustment", _
        "Miscellaneous Income", _
        "Margin Int Charged-Adj")

    a3 = Array( _
        "Margin Interest Charged", _
        "Policy Networked", _
        "Redemption", _
        "Transfer into Account", _
        "Transfer out of Account", _
        "Cashless Dividend", _
        "Exchange Deliver Out", _
        "Exchange Received In", _
        "Tax Exempt Income-Adj", _
        "Tax Withholding", _
        "Tax Withholding Adj", _
        "Tax Exempt Int Income-Adj", _
        "Pending Subscription")

    Dim total As Long
    total = (UBound(a1) + 1) + (UBound(a2) + 1) + (UBound(a3) + 1)

    Dim combined() As String
    ReDim combined(0 To total - 1)

    Dim idx As Long, i As Long
    idx = 0
    For i = 0 To UBound(a1): combined(idx) = CStr(a1(i)): idx = idx + 1: Next i
    For i = 0 To UBound(a2): combined(idx) = CStr(a2(i)): idx = idx + 1: Next i
    For i = 0 To UBound(a3): combined(idx) = CStr(a3(i)): idx = idx + 1: Next i

    DiscardActivityList = combined

End Function


' ================================================================
'  MAIN MACRO
' ================================================================
Sub ZelterMonthlyReportMacro()

    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    On Error GoTo ErrorHandler

    Dim wsSource As Worksheet
    Set wsSource = ActiveSheet

    ' ============================================================
    ' STEP 1 - Locate "Activity Date" header; delete all rows above
    ' ============================================================
    Dim hdrRow As Long
    hdrRow = 0
    Dim i As Long
    For i = 1 To 10000
        If Trim(CStr(wsSource.Cells(i, 1).Value)) = "Activity Date" Then
            hdrRow = i
            Exit For
        End If
    Next i

    If hdrRow = 0 Then
        MsgBox "ERROR: 'Activity Date' not found in column A. Macro aborted.", vbCritical
        GoTo Cleanup
    End If
    If hdrRow > 1 Then wsSource.Rows("1:" & hdrRow - 1).Delete

    ' ============================================================
    ' STEP 2 - Find first empty row in col A; delete it + below
    ' ============================================================
    Dim dLast As Long
    dLast = 1
    Do While Trim(CStr(wsSource.Cells(dLast + 1, 1).Value)) <> ""
        dLast = dLast + 1
        If dLast > 1000000 Then Exit Do
    Loop
    If dLast + 1 <= wsSource.Rows.Count Then
        wsSource.Rows(dLast + 1 & ":" & wsSource.Rows.Count).Delete
    End If

    ' ============================================================
    ' STEP 3 - Remove stale output tabs; create Spending Data and
    '          Discarded immediately so they always exist
    ' ============================================================
    Application.DisplayAlerts = False
    Dim staleNames(1 To 4) As String
    staleNames(1) = "Spending Data"
    staleNames(2) = "Discarded"
    staleNames(3) = "Deposits"
    staleNames(4) = "Withdrawals"

    Dim ws As Worksheet
    Dim si As Long
    For si = 1 To 4
        For Each ws In ThisWorkbook.Sheets
            If ws.Name = staleNames(si) Then
                ws.Delete
                Exit For
            End If
        Next ws
    Next si
    Application.DisplayAlerts = True

    ' Create Spending Data
    wsSource.Copy After:=wsSource
    Dim wsSD As Worksheet
    Set wsSD = ActiveSheet
    wsSD.Name = "Spending Data"

    ' Create Discarded immediately after Spending Data
    Dim wsDiscard As Worksheet
    Set wsDiscard = ThisWorkbook.Sheets.Add(After:=wsSD)
    wsDiscard.Name = "Discarded"
    wsSD.Rows(1).Copy wsDiscard.Rows(1)
    Dim discardNext As Long
    discardNext = 2

    ' ============================================================
    ' Resolve column indices from header row 1 of Spending Data
    ' ============================================================
    Dim nCols As Long
    nCols = wsSD.Cells(1, wsSD.Columns.Count).End(xlToLeft).Column

    Dim cActDate   As Long
    Dim cTransDate As Long
    Dim cAcct      As Long
    Dim cDesc      As Long
    Dim cActivity  As Long
    Dim cCheckNum  As Long
    Dim cCardNum   As Long
    Dim cSymbol    As Long
    Dim cCusip     As Long
    Dim cTags      As Long
    Dim cQty       As Long
    Dim cPrice     As Long
    Dim cAmt       As Long

    Dim col As Long
    For col = 1 To nCols
        Select Case Trim(CStr(wsSD.Cells(1, col).Value))
            Case "Activity Date":    cActDate   = col
            Case "Transaction Date": cTransDate = col
            Case "Account":          cAcct      = col
            Case "Description":      cDesc      = col
            Case "Activity":         cActivity  = col
            Case "Check Number":     cCheckNum  = col
            Case "Card Number":      cCardNum   = col
            Case "Symbol":           cSymbol    = col
            Case "Cusip":            cCusip     = col
            Case "Tags":             cTags      = col
            Case "Quantity":         cQty       = col
            Case "Price":            cPrice     = col
            Case "Amount($)":        cAmt       = col
        End Select
    Next col

    If cActivity = 0 Or cAcct = 0 Or cAmt = 0 Then
        MsgBox "ERROR: Required columns (Activity / Account / Amount($)) not found." & _
               vbCrLf & "Check that header names match exactly.", vbCritical
        GoTo Cleanup
    End If

    ' ============================================================
    ' STEP 4 - Apply AutoFilter to header row
    ' ============================================================
    wsSD.Rows(1).AutoFilter

    ' ============================================================
    ' STEP 5 - Move all qualifying activity rows to Discarded
    '          "Qualified Dividend" is intentionally skipped here
    ' ============================================================
    wsSD.AutoFilterMode = False

    Dim dList As Variant
    dList = DiscardActivityList()

    Dim rLast As Long
    rLast = wsSD.Cells(wsSD.Rows.Count, 1).End(xlUp).Row

    Dim r As Long
    Dim actStr As String
    Dim d As Long
    For r = rLast To 2 Step -1
        actStr = Trim(CStr(wsSD.Cells(r, cActivity).Value))
        If LCase(actStr) = "qualified dividend" Then GoTo Skip5
        For d = 0 To UBound(dList)
            If LCase(actStr) = LCase(CStr(dList(d))) Then
                wsSD.Rows(r).Copy wsDiscard.Rows(discardNext)
                discardNext = discardNext + 1
                wsSD.Rows(r).Delete
                Exit For
            End If
        Next d
        Skip5:
    Next r

    wsSD.Rows(1).AutoFilter

    ' ============================================================
    ' STEP 6 - Show only Qualified Dividend; discard Symbol = "APO"
    ' ============================================================
    wsSD.AutoFilterMode = False
    rLast = wsSD.Cells(wsSD.Rows.Count, 1).End(xlUp).Row

    wsSD.Range(wsSD.Cells(1, 1), wsSD.Cells(rLast, nCols)).AutoFilter _
        Field:=cActivity, Criteria1:="Qualified Dividend"

    For r = rLast To 2 Step -1
        If Not wsSD.Rows(r).Hidden Then
            If cSymbol > 0 Then
                If Trim(CStr(wsSD.Cells(r, cSymbol).Value)) = "APO" Then
                    wsSD.Rows(r).Copy wsDiscard.Rows(discardNext)
                    discardNext = discardNext + 1
                    wsSD.Rows(r).Delete
                End If
            End If
        End If
    Next r

    ' ============================================================
    ' STEP 7 - Show all records
    ' ============================================================
    wsSD.AutoFilterMode = False
    rLast = wsSD.Cells(wsSD.Rows.Count, 1).End(xlUp).Row
    wsSD.Range(wsSD.Cells(1, 1), wsSD.Cells(1, nCols)).AutoFilter

    ' ============================================================
    ' STEP 8 - Cash Transfer matching
    '
    '   Internal transfer pair criteria (all three must be true):
    '     a) Last 4 chars of Account(P) found in last line of Desc(Q)
    '     b) Last 4 chars of Account(Q) found in last line of Desc(P)
    '     c) Amount(P) + Amount(Q) = 0
    '   Matched pairs -> Discarded.  Unmatched -> stay.
    '   Greedy pairing prevents any row being matched twice.
    ' ============================================================
    wsSD.AutoFilterMode = False
    rLast = wsSD.Cells(wsSD.Rows.Count, 1).End(xlUp).Row

    wsSD.Range(wsSD.Cells(1, 1), wsSD.Cells(rLast, nCols)).AutoFilter _
        Field:=cActivity, Criteria1:="Cash Transfer"

    Dim ctRows() As Long
    Dim ctN As Long
    ctN = 0
    ReDim ctRows(1 To rLast)
    For r = 2 To rLast
        If Not wsSD.Rows(r).Hidden Then
            ctN = ctN + 1
            ctRows(ctN) = r
        End If
    Next r
    wsSD.AutoFilterMode = False

    If ctN >= 2 Then
        Dim ctMatched() As Boolean
        ReDim ctMatched(1 To ctN)

        Dim p As Long, q As Long
        Dim acctP As String, acctQ As String
        Dim last4P As String, last4Q As String
        Dim amtP As Double, amtQ As Double
        Dim descP As String, descQ As String
        Dim lastLineP As String, lastLineQ As String
        Dim linesP() As String, linesQ() As String
        Dim li As Long

        For p = 1 To ctN
            If Not ctMatched(p) Then
                acctP  = Trim(CStr(wsSD.Cells(ctRows(p), cAcct).Value))
                last4P = Right(acctP, 4)
                If Not IsNumeric(wsSD.Cells(ctRows(p), cAmt).Value) Then GoTo NextCT_P
                amtP = CDbl(wsSD.Cells(ctRows(p), cAmt).Value)

                descP = Replace(Replace(CStr(wsSD.Cells(ctRows(p), cDesc).Value), _
                                Chr(13) & Chr(10), Chr(10)), Chr(13), Chr(10))
                linesP = Split(descP, Chr(10))
                lastLineP = ""
                For li = UBound(linesP) To 0 Step -1
                    If Trim(linesP(li)) <> "" Then
                        lastLineP = Trim(linesP(li)): Exit For
                    End If
                Next li

                For q = p + 1 To ctN
                    If Not ctMatched(q) Then
                        acctQ  = Trim(CStr(wsSD.Cells(ctRows(q), cAcct).Value))
                        last4Q = Right(acctQ, 4)
                        If Not IsNumeric(wsSD.Cells(ctRows(q), cAmt).Value) Then GoTo NextCT_Q
                        amtQ = CDbl(wsSD.Cells(ctRows(q), cAmt).Value)

                        descQ = Replace(Replace(CStr(wsSD.Cells(ctRows(q), cDesc).Value), _
                                        Chr(13) & Chr(10), Chr(10)), Chr(13), Chr(10))
                        linesQ = Split(descQ, Chr(10))
                        lastLineQ = ""
                        For li = UBound(linesQ) To 0 Step -1
                            If Trim(linesQ(li)) <> "" Then
                                lastLineQ = Trim(linesQ(li)): Exit For
                            End If
                        Next li

                        If Len(last4P) > 0 And Len(last4Q) > 0 Then
                            If InStr(lastLineQ, last4P) > 0 And _
                               InStr(lastLineP, last4Q) > 0 And _
                               Abs(amtP + amtQ) < 0.01 Then
                                ctMatched(p) = True
                                ctMatched(q) = True
                                Exit For
                            End If
                        End If
                        NextCT_Q:
                    End If
                Next q
                NextCT_P:
            End If
        Next p

        Dim toDelete() As Long
        Dim delN As Long
        delN = 0
        ReDim toDelete(1 To ctN)
        For p = 1 To ctN
            If ctMatched(p) Then: delN = delN + 1: toDelete(delN) = ctRows(p)
        Next p

        Dim tmpRow As Long
        For p = 1 To delN - 1
            For q = p + 1 To delN
                If toDelete(p) < toDelete(q) Then
                    tmpRow = toDelete(p): toDelete(p) = toDelete(q): toDelete(q) = tmpRow
                End If
            Next q
        Next p

        For p = 1 To delN
            wsSD.Rows(toDelete(p)).Copy wsDiscard.Rows(discardNext)
            discardNext = discardNext + 1
            wsSD.Rows(toDelete(p)).Delete
        Next p
    End If

    ' ============================================================
    ' STEP 9 - Remove all filters
    ' ============================================================
    wsSD.AutoFilterMode = False

    ' ============================================================
    ' STEP 10 - Sort all data A-Z by Account
    ' ============================================================
    rLast = wsSD.Cells(wsSD.Rows.Count, 1).End(xlUp).Row
    wsSD.Range(wsSD.Cells(1, 1), wsSD.Cells(1, nCols)).AutoFilter

    With wsSD.Sort
        .SortFields.Clear
        .SortFields.Add Key:=wsSD.Columns(cAcct), _
                        SortOn:=xlSortOnValues, Order:=xlAscending
        .SetRange wsSD.Range(wsSD.Cells(1, 1), wsSD.Cells(rLast, nCols))
        .Header = xlYes
        .Apply
    End With

    ' ============================================================
    ' STEP 11 - Insert 3 separator rows between account groups
    '           and copy header directly above each new group.
    '
    '   Layout:
    '     [last row of Group A]
    '     [blank spacer]   <- ClearFormats removes inherited borders
    '     [blank spacer]   <- ClearFormats removes inherited borders
    '     [header copy]
    '     [first row of Group B]
    ' ============================================================
    wsSD.AutoFilterMode = False
    rLast = wsSD.Cells(wsSD.Rows.Count, 1).End(xlUp).Row

    Dim thisAcct As String, prevAcct As String
    For r = rLast To 3 Step -1
        thisAcct = Trim(CStr(wsSD.Cells(r, cAcct).Value))
        prevAcct = Trim(CStr(wsSD.Cells(r - 1, cAcct).Value))

        If thisAcct <> "" And prevAcct <> "" And thisAcct <> prevAcct Then
            wsSD.Rows(r).Resize(3).Insert Shift:=xlDown
            wsSD.Rows(r).ClearFormats       ' blank spacer 1
            wsSD.Rows(r + 1).ClearFormats   ' blank spacer 2
            wsSD.Rows(1).Copy wsSD.Rows(r + 2)  ' header copy above new group
        End If
    Next r

    ' ============================================================
    ' STEP 12 - Create Deposits and Withdrawals tabs
    ' ============================================================
    wsSD.Copy After:=wsSD
    Dim wsDeposits As Worksheet
    Set wsDeposits = ActiveSheet
    wsDeposits.Name = "Deposits"

    wsSD.Copy After:=wsDeposits
    Dim wsWithdrawals As Worksheet
    Set wsWithdrawals = ActiveSheet
    wsWithdrawals.Name = "Withdrawals"

    Call CleanAmountTab(wsDeposits, cAmt, True)
    Call CleanAmountTab(wsWithdrawals, cAmt, False)

    ' ------------------------------------------------------------------
    ' APO check for Deposits tab only.
    ' APO is a dividend stock - it produces positive (income) entries,
    ' so it can only appear on the Deposits side.  If APO is present on
    ' Deposits we keep Symbol, Cusip, Qty, and Price so that data is
    ' visible.  If APO is absent those columns are redundant and removed.
    '
    ' Withdrawals: Symbol, Cusip, Qty, and Price are ALWAYS removed
    ' because APO will never appear there (it generates income, not
    ' withdrawals).
    ' ------------------------------------------------------------------
    Dim hasAPO As Boolean
    hasAPO = False
    Dim depEnd As Long
    depEnd = wsDeposits.Cells(wsDeposits.Rows.Count, 1).End(xlUp).Row
    If cSymbol > 0 Then
        For r = 2 To depEnd
            If Trim(CStr(wsDeposits.Cells(r, cSymbol).Value)) = "APO" Then
                hasAPO = True: Exit For
            End If
        Next r
    End If

    ' Deposits:
    '   Always delete  -> Transaction Date, Check Number, Card Number, Tags
    '   Only if no APO -> Symbol, Cusip, Quantity, Price
    Call DeleteSpecifiedColumns(wsDeposits, _
                                cTransDate, cCheckNum, cCardNum, _
                                cSymbol, cCusip, cTags, cQty, cPrice, _
                                Not hasAPO, False)

    ' Withdrawals:
    '   Always delete  -> Transaction Date, Check Number, Card Number,
    '                     Tags, Symbol, Cusip, Quantity, Price
    '   No APO check needed - pass deleteSymbolGroup = True unconditionally
    Call DeleteSpecifiedColumns(wsWithdrawals, _
                                cTransDate, cCheckNum, cCardNum, _
                                cSymbol, cCusip, cTags, cQty, cPrice, _
                                True, False)

    ' ============================================================
    ' STEP 13 - Summary tables (SUMIF formulas, not hard-coded)
    ' ============================================================
    Call AddSummaryTable(wsDeposits)
    Call AddSummaryTable(wsWithdrawals)

    Application.ScreenUpdating = True
    Application.Calculation = xlCalculationAutomatic
    MsgBox "ZelterMonthlyReportMacro completed successfully!" & vbCrLf & _
           "Rows moved to Discarded: " & discardNext - 2, vbInformation
    Exit Sub

ErrorHandler:
    Application.ScreenUpdating = True
    Application.Calculation = xlCalculationAutomatic
    MsgBox "Error " & Err.Number & ": " & Err.Description, vbCritical
    Exit Sub

Cleanup:
    Application.ScreenUpdating = True
    Application.Calculation = xlCalculationAutomatic

End Sub


' ================================================================
'  HELPER : CleanAmountTab
'
'  keepPositive = True  -> Deposits    (delete rows where amt < 0)
'  keepPositive = False -> Withdrawals (delete rows where amt > 0)
'
'  Pass 1 : Remove data rows with wrong sign
'  Pass 2 : Remove orphan account blocks (header copy + no data)
'  Pass 3 : Remove any blank rows or stray header copies that end
'           up directly below original header row after orphan
'           removal (prevents duplicate title row at top)
' ================================================================
Private Sub CleanAmountTab(ws As Worksheet, cAmt As Long, keepPositive As Boolean)

    Dim r As Long, rLast As Long
    rLast = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row

    Dim cellA As String, amtV As Double, doDelete As Boolean
    For r = rLast To 2 Step -1
        cellA = Trim(CStr(ws.Cells(r, 1).Value))
        If cellA = "" Or cellA = "Activity Date" Then GoTo P1Next
        If Not IsNumeric(ws.Cells(r, cAmt).Value) Then GoTo P1Next

        amtV = CDbl(ws.Cells(r, cAmt).Value)
        doDelete = False
        If keepPositive     And amtV < 0 Then doDelete = True
        If Not keepPositive And amtV > 0 Then doDelete = True
        If doDelete Then ws.Rows(r).Delete
        P1Next:
    Next r

    rLast = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    Dim startBlock As Long, hasData As Boolean, fwd As Long, fwdVal As String

    For r = rLast To 2 Step -1
        If r > 1 And Trim(CStr(ws.Cells(r, 1).Value)) = "Activity Date" Then
            hasData = False
            For fwd = r + 1 To rLast
                fwdVal = Trim(CStr(ws.Cells(fwd, 1).Value))
                If fwdVal = "Activity Date" Then Exit For
                If fwdVal <> "" Then: hasData = True: Exit For
            Next fwd

            If Not hasData Then
                startBlock = r
                Do While startBlock > 2
                    If Trim(CStr(ws.Cells(startBlock - 1, 1).Value)) = "" Then
                        startBlock = startBlock - 1
                    Else
                        Exit Do
                    End If
                Loop
                ws.Rows(startBlock & ":" & r).Delete
                rLast = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
            End If
        End If
    Next r

    rLast = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    Do While rLast >= 2
        Dim row2Val As String
        row2Val = Trim(CStr(ws.Cells(2, 1).Value))
        If row2Val = "" Or row2Val = "Activity Date" Then
            ws.Rows(2).Delete
            rLast = rLast - 1
        Else
            Exit Do
        End If
    Loop

End Sub


' ================================================================
'  HELPER : DeleteSpecifiedColumns
'
'  Always deletes    : Transaction Date, Check Number, Card Number, Tags
'  deleteSymbolGroup : also deletes Symbol, Cusip, Quantity, Price
'  deletePriceAlways : (reserved / unused - Price is now fully
'                      governed by deleteSymbolGroup on both tabs)
'
'  All deletions happen right-to-left to keep column indices valid.
' ================================================================
Private Sub DeleteSpecifiedColumns(ws As Worksheet, _
    cTransDate As Long, cCheckNum As Long, cCardNum As Long, _
    cSymbol As Long, cCusip As Long, cTags As Long, _
    cQty As Long, cPrice As Long, _
    deleteSymbolGroup As Boolean, deletePriceAlways As Boolean)

    Dim colList(1 To 9) As Long
    Dim n As Long
    n = 0

    If cTransDate > 0 Then: n = n + 1: colList(n) = cTransDate
    If cCheckNum  > 0 Then: n = n + 1: colList(n) = cCheckNum
    If cCardNum   > 0 Then: n = n + 1: colList(n) = cCardNum
    If cTags      > 0 Then: n = n + 1: colList(n) = cTags

    If (deletePriceAlways Or deleteSymbolGroup) And cPrice > 0 Then
        n = n + 1: colList(n) = cPrice
    End If

    If deleteSymbolGroup Then
        If cSymbol > 0 Then: n = n + 1: colList(n) = cSymbol
        If cCusip  > 0 Then: n = n + 1: colList(n) = cCusip
        If cQty    > 0 Then: n = n + 1: colList(n) = cQty
    End If

    If n = 0 Then Exit Sub

    Dim a As Long, b As Long, tmp As Long
    For a = 1 To n - 1
        For b = a + 1 To n
            If colList(a) < colList(b) Then
                tmp = colList(a): colList(a) = colList(b): colList(b) = tmp
            End If
        Next b
    Next a

    For a = 1 To n
        If colList(a) > 0 Then ws.Columns(colList(a)).Delete
    Next a

End Sub


' ================================================================
'  HELPER : AddSummaryTable
'
'  Writes SUMIF formulas (not hard-coded values) so totals stay
'  live.  Range is bounded to data rows only (rows 2 : rLast) to
'  avoid any circular reference with the summary block itself.
'
'  Positioned 3 rows below last data row, in the two rightmost
'  columns.  Bold headers, full borders on the entire block.
' ================================================================
Private Sub AddSummaryTable(ws As Worksheet)

    Dim rLast As Long, cLast As Long
    rLast = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
    cLast = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column
    If rLast < 2 Or cLast < 2 Then Exit Sub

    Dim colAcct As Long, colAmt As Long, c As Long
    For c = 1 To cLast
        Select Case Trim(CStr(ws.Cells(1, c).Value))
            Case "Account":   colAcct = c
            Case "Amount($)": colAmt  = c
        End Select
    Next c
    If colAcct = 0 Or colAmt = 0 Then Exit Sub

    Dim accts(1 To 500) As String
    Dim n As Long
    n = 0
    Dim r As Long, v1 As String, acctName As String, found As Boolean, k As Long

    For r = 2 To rLast
        v1 = Trim(CStr(ws.Cells(r, 1).Value))
        If v1 = "" Or v1 = "Activity Date" Then GoTo SumNext
        If Not IsNumeric(ws.Cells(r, colAmt).Value) Then GoTo SumNext

        acctName = Trim(CStr(ws.Cells(r, colAcct).Value))
        If acctName = "" Or acctName = "Account" Then GoTo SumNext

        found = False
        For k = 1 To n
            If accts(k) = acctName Then: found = True: Exit For
        Next k
        If Not found Then: n = n + 1: accts(n) = acctName
        SumNext:
    Next r
    If n = 0 Then Exit Sub

    Dim startRow As Long, leftCol As Long, rightCol As Long
    startRow = rLast + 3
    leftCol  = cLast - 1
    rightCol = cLast

    Dim acctLetter As String, amtLetter As String
    acctLetter = Split(ws.Cells(1, colAcct).Address, "$")(1)
    amtLetter  = Split(ws.Cells(1, colAmt).Address, "$")(1)

    Dim acctRange As String, amtRange As String
    acctRange = "$" & acctLetter & "$2:$" & acctLetter & "$" & rLast
    amtRange  = "$" & amtLetter  & "$2:$" & amtLetter  & "$" & rLast

    ws.Cells(startRow, leftCol).Value  = "Account"
    ws.Cells(startRow, rightCol).Value = "Amount($)"
    ws.Cells(startRow, leftCol).Font.Bold  = True
    ws.Cells(startRow, rightCol).Font.Bold = True

    For k = 1 To n
        ws.Cells(startRow + k, leftCol).Value = accts(k)
        ws.Cells(startRow + k, rightCol).Formula = _
            "=SUMIF(" & acctRange & "," & _
            ws.Cells(startRow + k, leftCol).Address & "," & _
            amtRange & ")"
        ws.Cells(startRow + k, rightCol).NumberFormat = "#,##0.00"
    Next k

    Dim sumRng As Range
    Set sumRng = ws.Range( _
        ws.Cells(startRow, leftCol), _
        ws.Cells(startRow + n, rightCol))
    With sumRng.Borders
        .LineStyle = xlContinuous
        .Weight    = xlThin
        .Color     = RGB(0, 0, 0)
    End With

    ws.Columns(leftCol).AutoFit
    ws.Columns(rightCol).AutoFit

End Sub
