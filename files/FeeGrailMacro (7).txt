Option Explicit

'===============================================================
' Fee Grail Macro
'
'  1. Confirms layout, then strips the 5 rows above the real
'     header row (row 6, which must read "Contact")
'  2. Deletes the disclaimer block below the data outright:
'     scans column B (Account Number) for the first empty
'     cell, then deletes that row and everything below it -
'     done first, before anything else touches the sheet
'  3. Inserts a "Keep/Remove" column next to "Account Type",
'     pushing "Asset Allocation - Alternatives ($)" from G to H
'  4. Classifies every account type with a hard-coded formula
'     (no guessing) - blank Account Type is auto-REMOVE - and
'     HALTS if anything unrecognized shows up
'  5. Sorts so REMOVE rows group together, then moves them onto a
'     "Removed from Report" tab instead of deleting them, so
'     nothing is ever permanently lost and everything can be
'     traced back if a number looks wrong
'
'  Run this with the report open as the ActiveWorkbook and the
'  report tab as its ActiveSheet.
'===============================================================

Sub FeeGrailMacro()

    Dim wb As Workbook
    Dim ws As Worksheet
    Dim removeSheet As Worksheet
    Dim lastRow As Long, lastCol As Long, finalLastRow As Long
    Dim i As Long, destRow As Long
    Dim removeCount As Long
    Dim keepList As Variant, removeList As Variant
    Dim keepClause As String, removeClause As String, formulaText As String
    Dim flagFound As Boolean, flagValues As String
    Dim currentStep As String
    Const REMOVED_SHEET_NAME As String = "Removed from Report"

    Application.ScreenUpdating = False
    Application.EnableEvents = False
    On Error GoTo ErrHandler

    Set wb = ActiveWorkbook
    Set ws = wb.ActiveSheet

    ' A protected sheet blocks the insert/sort/delete steps below with a
    ' generic error, so check for it up front and give a clear message.
    currentStep = "checking sheet protection"
    If ws.ProtectContents Then
        MsgBox "This worksheet is protected, which blocks the insert/sort/delete steps this macro needs." & _
               vbNewLine & "Please unprotect it (Review tab > Unprotect Sheet) and re-run.", _
               vbCritical, "Fee Grail Macro - Sheet Protected"
        GoTo CleanExit
    End If

    '----------------------------------------------------------
    ' STEP 1 - Confirm layout, then delete the 5 rows above the
    ' real header row (row 6, which should read "Contact")
    '----------------------------------------------------------
    currentStep = "Step 1 (checking layout / deleting header rows)"
    If StrComp(Trim(ws.Range("A6").Value), "Contact", vbTextCompare) <> 0 Then
        MsgBox "Expected cell A6 to read 'Contact' but found '" & ws.Range("A6").Value & _
               "'. Macro stopped - please check the sheet layout before re-running.", _
               vbCritical, "Fee Grail Macro - Layout Mismatch"
        GoTo CleanExit
    End If
    ws.Rows("1:5").Delete

    '----------------------------------------------------------
    ' STEP 2 - Remove the disclaimer block below the data now,
    ' before anything else touches the sheet. Scan column B
    ' (Account Number) for the first empty cell, then delete
    ' that row and every row below it outright. Doing this
    ' first - before the Keep/Remove column, the sort, or any
    ' cut/paste - means those rows (and any merged cells inside
    ' them) are gone before any operation that could trip on
    ' them.
    '----------------------------------------------------------
    currentStep = "Step 2 (removing the disclaimer block below the data)"
    Dim emptyAcctRow As Long
    Dim foundCell As Range, lastUsedRow As Long

    Set foundCell = ws.Cells.Find(What:="*", After:=ws.Cells(1, 1), _
        LookIn:=xlValues, SearchOrder:=xlByRows, SearchDirection:=xlPrevious)
    If Not foundCell Is Nothing Then
        lastUsedRow = foundCell.Row
    Else
        lastUsedRow = ws.Cells(ws.Rows.Count, "A").End(xlUp).Row
    End If

    emptyAcctRow = 0
    For i = 2 To lastUsedRow
        If Trim(ws.Cells(i, "B").Value) = "" Then
            emptyAcctRow = i
            Exit For
        End If
    Next i

    If emptyAcctRow > 0 Then
        ws.Rows(emptyAcctRow & ":" & lastUsedRow).Delete
    End If

    '----------------------------------------------------------
    ' STEP 3 - Insert "Keep/Remove" as new column G, pushing the
    ' old column G ("Asset Allocation - Alternatives ($)") to H
    '----------------------------------------------------------
    currentStep = "Step 3 (inserting the Keep/Remove column)"
    If StrComp(Trim(ws.Range("F1").Value), "Account Type", vbTextCompare) <> 0 Then
        MsgBox "Expected cell F1 to read 'Account Type' but found '" & ws.Range("F1").Value & _
               "'. Macro stopped - please check the sheet layout before re-running.", _
               vbCritical, "Fee Grail Macro - Layout Mismatch"
        GoTo CleanExit
    End If

    ws.Columns("G:G").Insert Shift:=xlToRight
    ws.Range("G1").Value = "Keep/Remove"

    lastRow = ws.Cells(ws.Rows.Count, "F").End(xlUp).Row
    lastCol = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column

    If lastRow < 2 Then
        MsgBox "No data rows were found under 'Account Type'. Macro stopped.", vbExclamation, "Fee Grail Macro"
        GoTo CleanExit
    End If

    '----------------------------------------------------------
    ' STEP 4 - Build the KEEP / REMOVE formula from the two
    ' hard-coded lists below and fill it down column G
    '----------------------------------------------------------
    currentStep = "Step 4 (building/filling the Keep-Remove formula)"
    keepList = Array( _
        "Portfolio Management", _
        "Alternative Investments Advisory", _
        "Select UMA", _
        "Portfolio Management IRA", _
        "Consulting Group Advisor", _
        "Select UMA IRA", _
        "Portfolio Management RPM", _
        "Consulting and Evaluation Services", _
        "Alternative Investments Advisory IRA", _
        "Select UMA 529")

    removeList = Array( _
        "Alternative Investments Performance Reporting", _
        "AAA", _
        "Platinum CashPlus", _
        "IRA", _
        "REG", _
        "Premier CashPlus", _
        "RPM", _
        "529", _
        "VIP", _
        "LAL", _
        "TL", _
        "Investment Management Services")

    keepClause = BuildOrClause("F2", keepList)
    removeClause = BuildOrClause("F2", removeList)

    formulaText = "=IF(TRIM(F2)="""",""REMOVE"",IF(OR(" & keepClause & "),""KEEP"",IF(OR(" & removeClause & "),""REMOVE"",""FLAG"")))"

    ' Assign the formula directly to the whole range in one shot - Excel
    ' adjusts the relative F-column reference per row automatically, the
    ' same way a fill-down would, without going through Copy/Paste (which
    ' can conflict with table formatting/banding on some sheets)
    ws.Range("G2:G" & lastRow).Formula = formulaText

    ' Lock in the results as plain text so nothing shifts when we
    ' sort and move rows below
    ws.Range("G2:G" & lastRow).Value = ws.Range("G2:G" & lastRow).Value

    '----------------------------------------------------------
    ' STEP 5 - Safety check: stop cold if any account type did
    ' not match either list. Nothing gets sorted or moved yet.
    '----------------------------------------------------------
    currentStep = "Step 5 (scanning for unrecognized Account Types)"
    flagFound = False
    flagValues = ""
    For i = 2 To lastRow
        If ws.Cells(i, "G").Value = "FLAG" Then
            flagFound = True
            flagValues = flagValues & "- Row " & i & ":  """ & ws.Cells(i, "F").Value & """" & vbNewLine
        End If
    Next i

    If flagFound Then
        MsgBox "One or more Account Type values did not match the known Keep/Remove list." & vbNewLine & _
               "No rows were sorted or moved - review these and re-run:" & vbNewLine & vbNewLine & _
               flagValues, vbCritical, "Fee Grail Macro - Unrecognized Account Type"
        GoTo CleanExit
    End If

    '----------------------------------------------------------
    ' STEP 6 - Sort Z-to-A on Keep/Remove so REMOVE rows land
    ' together right under the header
    '----------------------------------------------------------
    currentStep = "Step 6 (sorting by Keep/Remove)"

    Dim sortRange As Range
    Set sortRange = ws.Range(ws.Cells(1, 1), ws.Cells(lastRow, lastCol))

    ' Excel's Sort requires every merged cell in the range to be the same
    ' size, and errors out (rather than just skipping) if even one cell in
    ' the range is merged differently from the rest. Scan the exact range
    ' we're about to sort so we can name the row(s) involved instead of
    ' guessing, then try a targeted unmerge scoped to just this range.
    Dim mergedRowsFound As String
    Dim r As Long, c As Long
    mergedRowsFound = ""
    For r = 1 To lastRow
        For c = 1 To lastCol
            If ws.Cells(r, c).MergeCells Then
                mergedRowsFound = mergedRowsFound & "- Row " & r & vbNewLine
                Exit For
            End If
        Next c
    Next r

    If Len(mergedRowsFound) > 0 Then
        On Error Resume Next
        Err.Clear
        sortRange.UnMerge
        If Err.Number <> 0 Then
            Err.Clear
            On Error GoTo ErrHandler
            MsgBox "Found merged cell(s) on the following row(s) within the data being sorted, " & _
                   "and an automatic unmerge attempt on them also failed:" & vbNewLine & vbNewLine & _
                   mergedRowsFound & vbNewLine & _
                   "Please open that row, unmerge those cells manually (Home tab > Merge & Center), " & _
                   "and re-run the macro.", vbCritical, "Fee Grail Macro - Merged Cells Found"
            GoTo CleanExit
        End If
        On Error GoTo ErrHandler
    End If

    With ws.Sort
        .SortFields.Clear
        .SortFields.Add Key:=ws.Range("G2:G" & lastRow), SortOn:=xlSortOnValues, Order:=xlDescending
        .SetRange sortRange
        .Header = xlYes
        .Apply
    End With

    '----------------------------------------------------------
    ' STEP 7 - Move REMOVE rows (plus one header row) to the
    ' "Removed from Report" tab, creating it if needed. Rows are
    ' CUT (moved), never deleted outright.
    '----------------------------------------------------------
    currentStep = "Step 7 (moving REMOVE rows to the tracking tab)"
    removeCount = Application.WorksheetFunction.CountIf(ws.Range("G2:G" & lastRow), "REMOVE")

    On Error Resume Next
    Set removeSheet = wb.Worksheets(REMOVED_SHEET_NAME)
    On Error GoTo ErrHandler
    If removeSheet Is Nothing Then
        Set removeSheet = wb.Worksheets.Add(After:=wb.Worksheets(wb.Worksheets.Count))
        removeSheet.Name = REMOVED_SHEET_NAME
    End If

    destRow = removeSheet.Cells(removeSheet.Rows.Count, 1).End(xlUp).Row
    If destRow = 1 And Len(removeSheet.Cells(1, 1).Value) = 0 Then
        ws.Range(ws.Cells(1, 1), ws.Cells(1, lastCol)).Copy Destination:=removeSheet.Range("A1")
        destRow = 1
    End If

    If removeCount > 0 Then
        On Error Resume Next
        Err.Clear
        ws.Range(ws.Cells(2, 1), ws.Cells(removeCount + 1, lastCol)).Cut _
            Destination:=removeSheet.Cells(destRow + 1, 1)
        If Err.Number <> 0 Then
            Dim cutErrDesc As String
            cutErrDesc = Err.Description
            Err.Clear
            On Error GoTo ErrHandler
            MsgBox "Could not move the REMOVE rows - one of them likely contains a merged cell " & _
                   "that didn't get cleared (" & cutErrDesc & ")." & vbNewLine & _
                   "Please check for merged cells among the account rows, unmerge them, and re-run.", _
                   vbCritical, "Fee Grail Macro - Merged Cell in Data"
            GoTo CleanExit
        End If
        On Error GoTo ErrHandler
        ws.Range(ws.Cells(2, 1), ws.Cells(removeCount + 1, lastCol)).EntireRow.Delete
    End If

    removeSheet.Columns.AutoFit

    ' Re-apply the filter dropdowns on the working tab now that the
    ' column layout and row count have both changed. This is cosmetic,
    ' so if it fails for any reason, don't let it block a run that
    ' otherwise completed successfully.
    currentStep = "final step (re-applying AutoFilter)"
    finalLastRow = ws.Cells(ws.Rows.Count, "F").End(xlUp).Row
    On Error Resume Next
    If ws.AutoFilterMode Then ws.AutoFilterMode = False
    ws.Range(ws.Cells(1, 1), ws.Cells(finalLastRow, lastCol)).AutoFilter
    On Error GoTo ErrHandler

    ws.Activate
    MsgBox "Fee Grail Macro complete." & vbNewLine & removeCount & " row(s) moved to '" & _
           REMOVED_SHEET_NAME & "'. Working tab now shows KEEP rows only.", vbInformation, "Fee Grail Macro"

CleanExit:
    Application.CutCopyMode = False
    Application.ScreenUpdating = True
    Application.EnableEvents = True
    Exit Sub

ErrHandler:
    MsgBox "An error occurred during " & currentStep & ":" & vbNewLine & vbNewLine & _
           "Error " & Err.Number & " - " & Err.Description, vbCritical, "Fee Grail Macro - Error"
    Resume CleanExit

End Sub

Private Function BuildOrClause(cellRef As String, values As Variant) As String
    Dim i As Long
    Dim parts() As String
    ReDim parts(LBound(values) To UBound(values))
    For i = LBound(values) To UBound(values)
        parts(i) = cellRef & "=""" & values(i) & """"
    Next i
    BuildOrClause = Join(parts, ",")
End Function
