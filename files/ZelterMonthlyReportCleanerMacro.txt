Option Explicit

' ================================================================
'  Z E L T E R   M O N T H L Y   R E P O R T   C L E A N E R
' ================================================================
'
'  Run AFTER ZelterMonthlyReportMacro has completed.
'  Requires "Deposits" and "Withdrawals" tabs to exist in wb.
'
' ================================================================
Sub ZelterMonthlyReportCleanerMacro()

    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    On Error GoTo ErrorHandler

    Dim wb As Workbook
    Set wb = ActiveSheet.Parent

    ' ============================================================
    ' STEP 1 - Verify source tabs exist
    ' ============================================================
    Dim wsDeposits    As Worksheet
    Dim wsWithdrawals As Worksheet

    On Error Resume Next
    Set wsDeposits    = wb.Sheets("Deposits")
    Set wsWithdrawals = wb.Sheets("Withdrawals")
    On Error GoTo ErrorHandler

    If wsDeposits Is Nothing Then
        MsgBox "ERROR: 'Deposits' tab not found." & vbCrLf & _
               "Run ZelterMonthlyReportMacro first.", vbCritical
        GoTo Cleanup
    End If
    If wsWithdrawals Is Nothing Then
        MsgBox "ERROR: 'Withdrawals' tab not found." & vbCrLf & _
               "Run ZelterMonthlyReportMacro first.", vbCritical
        GoTo Cleanup
    End If

    ' ============================================================
    ' STEP 2 - Delete stale CLEAN tabs if they exist
    ' ============================================================
    Application.DisplayAlerts = False
    Dim wsIdx As Long
    Dim wsName As String
    Dim deleteAgain As Boolean
    Do
        deleteAgain = False
        For wsIdx = wb.Sheets.Count To 1 Step -1
            wsName = wb.Sheets(wsIdx).Name
            If wsName = "Deposits CLEAN" Or wsName = "Withdrawals CLEAN" Then
                wb.Sheets(wsIdx).Delete
                deleteAgain = True
                Exit For
            End If
        Next wsIdx
    Loop While deleteAgain
    Application.DisplayAlerts = True

    ' ============================================================
    ' STEP 3 - Clone originals into CLEAN tabs
    ' ============================================================
    wsDeposits.Copy After:=wsDeposits
    Dim wsDepClean As Worksheet
    Set wsDepClean = wb.Sheets(wsDeposits.Index + 1)
    wsDepClean.Name = "Deposits CLEAN"

    wsWithdrawals.Copy After:=wsWithdrawals
    Dim wsWdwClean As Worksheet
    Set wsWdwClean = wb.Sheets(wsWithdrawals.Index + 1)
    wsWdwClean.Name = "Withdrawals CLEAN"

    ' ============================================================
    ' STEP 4 - Clean descriptions
    ' ============================================================
    Call CleanDescriptions(wsDepClean)
    Call CleanDescriptions(wsWdwClean)

    ' ============================================================
    ' STEP 5 - Fix row heights
    ' ============================================================
    Call NormaliseRowHeights(wsDepClean)
    Call NormaliseRowHeights(wsWdwClean)

    ' ============================================================
    ' STEP 6 - Format summary boxes
    ' ============================================================
    Call FormatSummaryBox(wsDepClean)
    Call FormatSummaryBox(wsWdwClean)

    Application.ScreenUpdating = True
    Application.Calculation = xlCalculationAutomatic

    MsgBox "Cheers Beast," & vbCrLf & vbCrLf & _
           "Please check discarded Cash Transfers net to 0, " & _
           "and check any of their kept descriptions.", vbInformation
    Exit Sub

ErrorHandler:
    Application.ScreenUpdating = True
    Application.Calculation = xlCalculationAutomatic
    MsgBox "Error " & Err.Number & ": " & Err.Description, vbCritical
    Exit Sub

Cleanup:
    Application.ScreenUpdating = True
    Application.Calculation = xlCalculationAutomatic

End Sub


' ================================================================
'  HELPER : CleanDescriptions
' ================================================================
Private Sub CleanDescriptions(ws As Worksheet)

    Dim cActivity    As Long
    Dim cDescription As Long
    Dim cInstitution As Long
    Dim cLast        As Long
    Dim col          As Long

    cLast = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column
    For col = 1 To cLast
        Select Case Trim(CStr(ws.Cells(1, col).Value))
            Case "Activity":         cActivity    = col
            Case "Description":      cDescription = col
            Case "Institution Name": cInstitution = col
        End Select
    Next col

    If cActivity = 0 Or cDescription = 0 Then
        MsgBox "WARNING: 'Activity' or 'Description' column not found on '" & _
               ws.Name & "'. Skipping.", vbExclamation
        Exit Sub
    End If

    Dim rLast As Long
    rLast = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row

    Dim r           As Long
    Dim actVal      As String
    Dim instVal     As String
    Dim rawCell     As String
    Dim cleaned     As String
    Dim allLines()  As String
    Dim nonEmpty()  As String
    Dim lineCount   As Long
    Dim li          As Long
    Dim firstWord   As String

    For r = 2 To rLast

        actVal = Trim(CStr(ws.Cells(r, cActivity).Value))
        If actVal = "" Or actVal = "Activity" Then GoTo NextRow

        Select Case actVal
            Case "Funds Transferred", "Check", "Automated Payment", _
                 "Funds Paid", "Debit Card", "Withdrawal", "DEP", _
                 "DEPOSIT", "Direct Deposit", "Online Transfer", _
                 "Zelle Payment", "Funds Received"
            Case Else
                GoTo NextRow
        End Select

        If cInstitution > 0 Then
            instVal = Trim(CStr(ws.Cells(r, cInstitution).Value))
        Else
            instVal = ""
        End If

        rawCell = CStr(ws.Cells(r, cDescription).Value)
        rawCell = Replace(rawCell, Chr(13) & Chr(10), Chr(10))
        rawCell = Replace(rawCell, Chr(13), Chr(10))

        allLines = Split(rawCell, Chr(10))
        ReDim nonEmpty(0 To UBound(allLines))
        lineCount = 0
        For li = 0 To UBound(allLines)
            If Trim(allLines(li)) <> "" Then
                nonEmpty(lineCount) = Trim(allLines(li))
                lineCount = lineCount + 1
            End If
        Next li

        cleaned = ""

        Select Case actVal

            Case "Funds Transferred"
                If lineCount >= 2 Then
                    cleaned = nonEmpty(0) & " " & nonEmpty(1)
                Else
                    cleaned = nonEmpty(0)
                End If

            Case "Check"
                cleaned = nonEmpty(0)

            Case "Automated Payment"
                If lineCount <= 1 Then
                    cleaned = nonEmpty(0)
                ElseIf lineCount = 2 Then
                    cleaned = nonEmpty(0)
                Else
                    cleaned = nonEmpty(2)
                End If

            Case "Funds Paid"
                If lineCount = 1 Then
                    cleaned = nonEmpty(0)
                Else
                    cleaned = nonEmpty(1)
                    If Len(cleaned) > 5 Then
                        cleaned = Mid(cleaned, 6)
                    Else
                        cleaned = ""
                    End If
                End If

            Case "Debit Card"
                cleaned = nonEmpty(0)

            Case "Withdrawal"
                If lineCount = 1 Then
                    cleaned = nonEmpty(0)
                Else
                    If InStr(nonEmpty(0), " ") > 0 Then
                        firstWord = Left(nonEmpty(0), InStr(nonEmpty(0), " ") - 1)
                    Else
                        firstWord = nonEmpty(0)
                    End If
                    If LCase(firstWord) = "zelle" Then
                        cleaned = nonEmpty(0) & " " & nonEmpty(1)
                    Else
                        cleaned = nonEmpty(1)
                    End If
                End If

            Case "DEP"
                If lineCount = 1 Then
                    cleaned = nonEmpty(0)
                Else
                    If InStr(nonEmpty(0), " ") > 0 Then
                        firstWord = Left(nonEmpty(0), InStr(nonEmpty(0), " ") - 1)
                    Else
                        firstWord = nonEmpty(0)
                    End If
                    If LCase(firstWord) = "online" Then
                        cleaned = nonEmpty(0)
                    Else
                        cleaned = nonEmpty(0)
                    End If
                End If

            Case "DEPOSIT"
                If LCase(instVal) = LCase("Bank of America") Then
                    If lineCount >= 2 Then
                        Dim line2BofA As String
                        line2BofA = nonEmpty(1)
                        If Len(line2BofA) > 6 Then
                            line2BofA = Mid(line2BofA, 7)
                        Else
                            line2BofA = ""
                        End If
                        cleaned = nonEmpty(0) & " " & line2BofA
                    Else
                        cleaned = nonEmpty(0)
                    End If
                Else
                    GoTo NextRow
                End If

            Case "Direct Deposit"
                If LCase(instVal) = LCase("Morgan Stanley") Then
                    If lineCount >= 3 Then
                        cleaned = nonEmpty(1) & " " & nonEmpty(2)
                    ElseIf lineCount = 2 Then
                        cleaned = nonEmpty(1)
                    Else
                        cleaned = nonEmpty(0)
                    End If
                Else
                    GoTo NextRow
                End If

            Case "Online Transfer"
                If LCase(instVal) = LCase("Morgan Stanley") Then
                    If lineCount >= 2 Then
                        cleaned = nonEmpty(1)
                    Else
                        cleaned = nonEmpty(0)
                    End If
                Else
                    GoTo NextRow
                End If

            Case "Zelle Payment"
                If LCase(instVal) = LCase("Morgan Stanley") Then
                    cleaned = nonEmpty(0)
                Else
                    GoTo NextRow
                End If

            Case "Funds Received"
                If lineCount >= 3 Then
                    cleaned = nonEmpty(2)
                ElseIf lineCount = 2 Then
                    cleaned = nonEmpty(1)
                Else
                    cleaned = nonEmpty(0)
                End If

        End Select

        ws.Cells(r, cDescription).Value = cleaned

        NextRow:
    Next r

End Sub


' ================================================================
'  HELPER : NormaliseRowHeights
' ================================================================
Private Sub NormaliseRowHeights(ws As Worksheet)
    ws.UsedRange.WrapText = False
    ws.UsedRange.EntireRow.AutoFit
End Sub


' ================================================================
'  HELPER : FormatSummaryBox
'
'  FIX: previously used ws.Cells(ws.Rows.Count, 1).End(xlUp).Row
'  to find rLast, which stops at the last row of column 1 data.
'  The summary box lives in cLast-1 and cLast (different columns),
'  several rows below the last column-1 entry, so the scan never
'  reached it.
'
'  Fix: derive rLastLeft from the summary column (leftCol) itself
'  so the scan correctly covers the full extent of that column.
' ================================================================
Private Sub FormatSummaryBox(ws As Worksheet)

    Dim cLast As Long
    cLast = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column
    If cLast < 2 Then Exit Sub

    Dim leftCol  As Long
    Dim rightCol As Long
    leftCol  = cLast - 1
    rightCol = cLast

    ' Scan from the bottom of the LEFT SUMMARY COLUMN (not col 1)
    ' upward to find the bold "Account" header of the summary box
    Dim rLastLeft As Long
    rLastLeft = ws.Cells(ws.Rows.Count, leftCol).End(xlUp).Row

    Dim summaryHeaderRow As Long
    summaryHeaderRow = 0
    Dim r As Long
    For r = rLastLeft To 1 Step -1
        If Trim(CStr(ws.Cells(r, leftCol).Value)) = "Account" And _
           ws.Cells(r, leftCol).Font.Bold = True Then
            summaryHeaderRow = r
            Exit For
        End If
    Next r

    If summaryHeaderRow = 0 Then Exit Sub   ' no summary box found

    ' Walk down to find the last row of the summary box
    Dim summaryLastRow As Long
    summaryLastRow = summaryHeaderRow
    For r = summaryHeaderRow + 1 To summaryHeaderRow + 200
        If Trim(CStr(ws.Cells(r, leftCol).Value)) = "" And _
           Trim(CStr(ws.Cells(r, rightCol).Value)) = "" Then
            Exit For
        End If
        summaryLastRow = r
    Next r

    ' Full summary range
    Dim sumRng As Range
    Set sumRng = ws.Range(ws.Cells(summaryHeaderRow, leftCol), _
                          ws.Cells(summaryLastRow, rightCol))

    ' Center horizontally + middle vertically on all cells
    sumRng.HorizontalAlignment = xlCenter
    sumRng.VerticalAlignment   = xlVAlignCenter

    ' Zero decimal places on the amount column (skip header row)
    Dim amtRng As Range
    Set amtRng = ws.Range(ws.Cells(summaryHeaderRow + 1, rightCol), _
                          ws.Cells(summaryLastRow, rightCol))
    amtRng.NumberFormat = "#,##0"

    ' Rename "Total($)" -> "Total" in the left column
    For r = summaryHeaderRow To summaryLastRow
        If Trim(CStr(ws.Cells(r, leftCol).Value)) = "Total($)" Then
            ws.Cells(r, leftCol).Value = "Total"
        End If
    Next r

End Sub
