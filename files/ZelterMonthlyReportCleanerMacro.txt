Option Explicit

' ================================================================
'  Z E L T E R   M O N T H L Y   R E P O R T   C L E A N E R
' ================================================================
'
'  Run AFTER ZelterMonthlyReportMacro has completed.
'  Requires "Deposits" and "Withdrawals" tabs to exist in the
'  active workbook.
'
'  WHAT THIS MACRO DOES:
'  1. Clones Deposits   -> "Deposits CLEAN"
'  2. Clones Withdrawals -> "Withdrawals CLEAN"
'  3. Cleans the Description column on each CLEAN tab based on
'     the Activity value for that row, according to these rules:
'
'  Activity = "Funds Transferred"
'    3-line cell: merge line1 + " " + line2, discard line3
'    Result: single line
'
'  Activity = "Check"
'    Keep line 1 only, discard all others
'
'  Activity = "Automated Payment"
'    2-line cell: keep line 1 only
'    3-line cell: keep line 3 only
'
'  Activity = "Funds Paid"
'    1-line cell: keep as-is
'    3-line cell: keep line 2, then strip first 5 characters
'
'  Activity = "Debit Card"
'    Keep line 1 only, discard all others
'
'  Activity = "Withdrawal"
'    1-line cell: keep as-is
'    2-line cell, first word of line1 = "Zelle":
'                 merge -> "line1 line2" (single line)
'    2-line cell, first word of line1 != "Zelle":
'                 keep line 2 only
'
'  All other Activity values: Description left untouched.
'
' ================================================================
Sub ZelterMonthlyReportCleanerMacro()

    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    On Error GoTo ErrorHandler

    ' ---------------------------------------------------------------
    ' Derive workbook from the active sheet so this works whether the
    ' macro lives in Personal.xlsb, an add-in, or the data file itself
    ' ---------------------------------------------------------------
    Dim wb As Workbook
    Set wb = ActiveSheet.Parent

    ' ============================================================
    ' STEP 1 - Verify source tabs exist
    ' ============================================================
    Dim wsDeposits    As Worksheet
    Dim wsWithdrawals As Worksheet

    On Error Resume Next
    Set wsDeposits    = wb.Sheets("Deposits")
    Set wsWithdrawals = wb.Sheets("Withdrawals")
    On Error GoTo ErrorHandler

    If wsDeposits Is Nothing Then
        MsgBox "ERROR: 'Deposits' tab not found." & vbCrLf & _
               "Run ZelterMonthlyReportMacro first.", vbCritical
        GoTo Cleanup
    End If
    If wsWithdrawals Is Nothing Then
        MsgBox "ERROR: 'Withdrawals' tab not found." & vbCrLf & _
               "Run ZelterMonthlyReportMacro first.", vbCritical
        GoTo Cleanup
    End If

    ' ============================================================
    ' STEP 2 - Delete stale CLEAN tabs if they exist
    ' ============================================================
    Application.DisplayAlerts = False
    Dim wsIdx As Long
    Dim wsName As String
    Dim deleteAgain As Boolean
    Do
        deleteAgain = False
        For wsIdx = wb.Sheets.Count To 1 Step -1
            wsName = wb.Sheets(wsIdx).Name
            If wsName = "Deposits CLEAN" Or wsName = "Withdrawals CLEAN" Then
                wb.Sheets(wsIdx).Delete
                deleteAgain = True
                Exit For
            End If
        Next wsIdx
    Loop While deleteAgain
    Application.DisplayAlerts = True

    ' ============================================================
    ' STEP 3 - Clone Deposits -> "Deposits CLEAN"
    '          Clone Withdrawals -> "Withdrawals CLEAN"
    '
    '  Copy After the originals so order is:
    '  ... | Deposits | Deposits CLEAN | Withdrawals | Withdrawals CLEAN | Discarded
    ' ============================================================
    wsDeposits.Copy After:=wsDeposits
    Dim wsDepClean As Worksheet
    Set wsDepClean = wb.Sheets(wsDeposits.Index + 1)
    wsDepClean.Name = "Deposits CLEAN"

    wsWithdrawals.Copy After:=wsWithdrawals
    Dim wsWdwClean As Worksheet
    Set wsWdwClean = wb.Sheets(wsWithdrawals.Index + 1)
    wsWdwClean.Name = "Withdrawals CLEAN"

    ' ============================================================
    ' STEP 4 - Clean both CLEAN tabs
    ' ============================================================
    Call CleanDescriptions(wsDepClean)
    Call CleanDescriptions(wsWdwClean)

    Application.ScreenUpdating = True
    Application.Calculation = xlCalculationAutomatic
    MsgBox "ZelterMonthlyReportCleanerMacro completed successfully!", vbInformation
    Exit Sub

ErrorHandler:
    Application.ScreenUpdating = True
    Application.Calculation = xlCalculationAutomatic
    MsgBox "Error " & Err.Number & ": " & Err.Description, vbCritical
    Exit Sub

Cleanup:
    Application.ScreenUpdating = True
    Application.Calculation = xlCalculationAutomatic

End Sub


' ================================================================
'  HELPER : CleanDescriptions
'
'  Iterates every data row on the given sheet.
'  Resolves Activity and Description column indices from the
'  header row so it is not dependent on column position.
'  Applies the appropriate cleaning rule based on Activity value.
' ================================================================
Private Sub CleanDescriptions(ws As Worksheet)

    ' --- Locate Activity and Description columns ---
    Dim cActivity    As Long
    Dim cDescription As Long
    Dim cLast        As Long
    Dim col          As Long

    cLast = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column
    For col = 1 To cLast
        Select Case Trim(CStr(ws.Cells(1, col).Value))
            Case "Activity":    cActivity    = col
            Case "Description": cDescription = col
        End Select
    Next col

    If cActivity = 0 Or cDescription = 0 Then
        MsgBox "WARNING: 'Activity' or 'Description' column not found on sheet '" & _
               ws.Name & "'. Skipping.", vbExclamation
        Exit Sub
    End If

    Dim rLast As Long
    rLast = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row

    Dim r As Long
    Dim actVal  As String
    Dim rawCell As String
    Dim cleaned As String
    Dim lines() As String
    Dim lineCount As Long

    For r = 2 To rLast

        actVal = Trim(CStr(ws.Cells(r, cActivity).Value))

        ' Skip blank rows and header-copy rows
        If actVal = "" Or actVal = "Activity" Then GoTo NextRow

        ' Only process rows whose Activity has a cleaning rule
        Select Case actVal
            Case "Funds Transferred", "Check", "Automated Payment", _
                 "Funds Paid", "Debit Card", "Withdrawal"
                ' fall through to cleaning logic below
            Case Else
                GoTo NextRow
        End Select

        ' Read raw cell value; normalise all line-break styles to Chr(10)
        rawCell = CStr(ws.Cells(r, cDescription).Value)
        rawCell = Replace(rawCell, Chr(13) & Chr(10), Chr(10))
        rawCell = Replace(rawCell, Chr(13), Chr(10))

        ' Split into lines and collect only non-empty ones
        Dim allLines() As String
        allLines = Split(rawCell, Chr(10))

        Dim nonEmpty() As String
        ReDim nonEmpty(0 To UBound(allLines))
        lineCount = 0
        Dim li As Long
        For li = 0 To UBound(allLines)
            If Trim(allLines(li)) <> "" Then
                nonEmpty(lineCount) = Trim(allLines(li))
                lineCount = lineCount + 1
            End If
        Next li
        ' lineCount now = number of meaningful lines (1-based count)

        ' ----------------------------------------------------------
        ' Apply cleaning rule by Activity
        ' ----------------------------------------------------------
        Select Case actVal

            ' ------------------------------------------------------
            ' FUNDS TRANSFERRED
            '   3 lines: merge line1 + " " + line2, drop line3
            '   Result : single line
            ' ------------------------------------------------------
            Case "Funds Transferred"
                If lineCount >= 2 Then
                    cleaned = nonEmpty(0) & " " & nonEmpty(1)
                Else
                    cleaned = nonEmpty(0)
                End If

            ' ------------------------------------------------------
            ' CHECK
            '   Keep line 1 only
            ' ------------------------------------------------------
            Case "Check"
                cleaned = nonEmpty(0)

            ' ------------------------------------------------------
            ' AUTOMATED PAYMENT
            '   2 lines -> keep line 1
            '   3 lines -> keep line 3
            ' ------------------------------------------------------
            Case "Automated Payment"
                If lineCount = 1 Then
                    cleaned = nonEmpty(0)
                ElseIf lineCount = 2 Then
                    cleaned = nonEmpty(0)   ' 2-line: keep line 1
                Else
                    cleaned = nonEmpty(2)   ' 3-line: keep line 3 (index 2)
                End If

            ' ------------------------------------------------------
            ' FUNDS PAID
            '   1 line  -> keep as-is
            '   3 lines -> keep line 2, then strip first 5 chars
            ' ------------------------------------------------------
            Case "Funds Paid"
                If lineCount = 1 Then
                    cleaned = nonEmpty(0)
                Else
                    ' Use line 2 (index 1) regardless of total count
                    cleaned = nonEmpty(1)
                    ' Strip first 5 characters
                    If Len(cleaned) > 5 Then
                        cleaned = Mid(cleaned, 6)
                    Else
                        cleaned = ""
                    End If
                End If

            ' ------------------------------------------------------
            ' DEBIT CARD
            '   Keep line 1 only
            ' ------------------------------------------------------
            Case "Debit Card"
                cleaned = nonEmpty(0)

            ' ------------------------------------------------------
            ' WITHDRAWAL
            '   S1 (2 lines, line1 starts with "Zelle"):
            '      merge -> "line1 line2"
            '   S2 (2 lines, line1 does NOT start with "Zelle"):
            '      keep line 2 only
            '   S3 (1 line):
            '      keep as-is
            ' ------------------------------------------------------
            Case "Withdrawal"
                If lineCount = 1 Then
                    cleaned = nonEmpty(0)
                ElseIf lineCount >= 2 Then
                    Dim firstWord As String
                    firstWord = Split(Trim(nonEmpty(0)), " ")(0)
                    If LCase(firstWord) = "zelle" Then
                        cleaned = nonEmpty(0) & " " & nonEmpty(1)  ' S1: merge
                    Else
                        cleaned = nonEmpty(1)                        ' S2: line 2 only
                    End If
                End If

        End Select

        ' Write the cleaned single-line value back to the cell
        ws.Cells(r, cDescription).Value = cleaned

        NextRow:
    Next r

End Sub
