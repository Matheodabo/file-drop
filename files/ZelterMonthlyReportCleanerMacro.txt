Option Explicit

' ================================================================
'  Z E L T E R   M O N T H L Y   R E P O R T   C L E A N E R
' ================================================================
'
'  Run AFTER ZelterMonthlyReportMacro has completed.
'  Requires "Deposits" and "Withdrawals" tabs to exist in wb.
'
'  CLEANING RULES (Description column, based on Activity + Institution):
'
'  "Funds Transferred"   3 lines  -> merge line1 + " " + line2, drop line3
'  "Check"                        -> keep line 1 only
'  "Automated Payment"   2 lines  -> keep line 1
'                        3 lines  -> keep line 3
'  "Funds Paid"          1 line   -> keep as-is
'                        3 lines  -> keep line 2, strip first 5 chars
'  "Debit Card"                   -> keep line 1 only
'  "Withdrawal"          1 line   -> keep as-is
'                        2 lines, line1 starts "Zelle" -> merge line1 + " " + line2
'                        2 lines, line1 NOT "Zelle"    -> keep line 2 only
'  "DEP"                 2 lines, line1 starts "Online" -> keep line 1 only
'                        1 line   -> keep as-is
'  "DEPOSIT"             Institution = "Bank of America"
'                        3 lines  -> strip first 6 chars of line2,
'                                    drop line3, merge line1 + " " + line2
'  "Direct Deposit"      Institution = "Morgan Stanley"
'                        3 lines  -> drop line1, merge line2 + " " + line3
'  "Online Transfer"     Institution = "Morgan Stanley"
'                        3 lines  -> keep line 2 only
'  "Zelle Payment"       Institution = "Morgan Stanley"
'                        2 lines  -> keep line 1 only
'  "Funds Received"      3 lines  -> keep line 3 only
'
'  After all cleaning: WrapText disabled + rows AutoFit on both CLEAN tabs
'
' ================================================================
Sub ZelterMonthlyReportCleanerMacro()

    Application.ScreenUpdating = False
    Application.Calculation = xlCalculationManual
    On Error GoTo ErrorHandler

    ' Derive workbook from active sheet - safe whether macro lives in
    ' Personal.xlsb, an add-in, or the data file itself
    Dim wb As Workbook
    Set wb = ActiveSheet.Parent

    ' ============================================================
    ' STEP 1 - Verify source tabs exist
    ' ============================================================
    Dim wsDeposits    As Worksheet
    Dim wsWithdrawals As Worksheet

    On Error Resume Next
    Set wsDeposits    = wb.Sheets("Deposits")
    Set wsWithdrawals = wb.Sheets("Withdrawals")
    On Error GoTo ErrorHandler

    If wsDeposits Is Nothing Then
        MsgBox "ERROR: 'Deposits' tab not found." & vbCrLf & _
               "Run ZelterMonthlyReportMacro first.", vbCritical
        GoTo Cleanup
    End If
    If wsWithdrawals Is Nothing Then
        MsgBox "ERROR: 'Withdrawals' tab not found." & vbCrLf & _
               "Run ZelterMonthlyReportMacro first.", vbCritical
        GoTo Cleanup
    End If

    ' ============================================================
    ' STEP 2 - Delete stale CLEAN tabs if they exist
    ' ============================================================
    Application.DisplayAlerts = False
    Dim wsIdx As Long
    Dim wsName As String
    Dim deleteAgain As Boolean
    Do
        deleteAgain = False
        For wsIdx = wb.Sheets.Count To 1 Step -1
            wsName = wb.Sheets(wsIdx).Name
            If wsName = "Deposits CLEAN" Or wsName = "Withdrawals CLEAN" Then
                wb.Sheets(wsIdx).Delete
                deleteAgain = True
                Exit For
            End If
        Next wsIdx
    Loop While deleteAgain
    Application.DisplayAlerts = True

    ' ============================================================
    ' STEP 3 - Clone originals into CLEAN tabs
    ' ============================================================
    wsDeposits.Copy After:=wsDeposits
    Dim wsDepClean As Worksheet
    Set wsDepClean = wb.Sheets(wsDeposits.Index + 1)
    wsDepClean.Name = "Deposits CLEAN"

    wsWithdrawals.Copy After:=wsWithdrawals
    Dim wsWdwClean As Worksheet
    Set wsWdwClean = wb.Sheets(wsWithdrawals.Index + 1)
    wsWdwClean.Name = "Withdrawals CLEAN"

    ' ============================================================
    ' STEP 4 - Clean descriptions on both CLEAN tabs
    ' ============================================================
    Call CleanDescriptions(wsDepClean)
    Call CleanDescriptions(wsWdwClean)

    ' ============================================================
    ' STEP 5 - Fix row heights on both CLEAN tabs
    '
    '  Multi-line cell content + WrapText causes rows to stay tall
    '  even after the value is replaced with a single line.
    '  Fix: disable WrapText on all cells, then AutoFit all rows.
    '  This replicates the behaviour of double-clicking a row border.
    ' ============================================================
    Call NormaliseRowHeights(wsDepClean)
    Call NormaliseRowHeights(wsWdwClean)

    Application.ScreenUpdating = True
    Application.Calculation = xlCalculationAutomatic
    MsgBox "ZelterMonthlyReportCleanerMacro completed successfully!", vbInformation
    Exit Sub

ErrorHandler:
    Application.ScreenUpdating = True
    Application.Calculation = xlCalculationAutomatic
    MsgBox "Error " & Err.Number & ": " & Err.Description, vbCritical
    Exit Sub

Cleanup:
    Application.ScreenUpdating = True
    Application.Calculation = xlCalculationAutomatic

End Sub


' ================================================================
'  HELPER : CleanDescriptions
'
'  Resolves Activity, Description, and Institution Name columns
'  by header name (position-independent).
'  Applies the appropriate rule to each data row.
' ================================================================
Private Sub CleanDescriptions(ws As Worksheet)

    ' --- Resolve column indices ---
    Dim cActivity     As Long
    Dim cDescription  As Long
    Dim cInstitution  As Long
    Dim cLast         As Long
    Dim col           As Long

    cLast = ws.Cells(1, ws.Columns.Count).End(xlToLeft).Column
    For col = 1 To cLast
        Select Case Trim(CStr(ws.Cells(1, col).Value))
            Case "Activity":         cActivity    = col
            Case "Description":      cDescription = col
            Case "Institution Name": cInstitution = col
        End Select
    Next col

    If cActivity = 0 Or cDescription = 0 Then
        MsgBox "WARNING: 'Activity' or 'Description' column not found on '" & _
               ws.Name & "'. Skipping.", vbExclamation
        Exit Sub
    End If

    Dim rLast As Long
    rLast = ws.Cells(ws.Rows.Count, 1).End(xlUp).Row

    Dim r            As Long
    Dim actVal       As String
    Dim instVal      As String
    Dim rawCell      As String
    Dim cleaned      As String
    Dim allLines()   As String
    Dim nonEmpty()   As String
    Dim lineCount    As Long
    Dim li           As Long
    Dim firstWord    As String

    For r = 2 To rLast

        actVal = Trim(CStr(ws.Cells(r, cActivity).Value))

        ' Skip blank rows and header-copy rows
        If actVal = "" Or actVal = "Activity" Then GoTo NextRow

        ' Only process rows whose Activity has a cleaning rule
        Select Case actVal
            Case "Funds Transferred", "Check", "Automated Payment", _
                 "Funds Paid", "Debit Card", "Withdrawal", "DEP", _
                 "DEPOSIT", "Direct Deposit", "Online Transfer", _
                 "Zelle Payment", "Funds Received"
                ' valid - fall through
            Case Else
                GoTo NextRow
        End Select

        ' Read Institution Name if the column exists
        If cInstitution > 0 Then
            instVal = Trim(CStr(ws.Cells(r, cInstitution).Value))
        Else
            instVal = ""
        End If

        ' Normalise line breaks and collect non-empty lines
        rawCell = CStr(ws.Cells(r, cDescription).Value)
        rawCell = Replace(rawCell, Chr(13) & Chr(10), Chr(10))
        rawCell = Replace(rawCell, Chr(13), Chr(10))

        allLines = Split(rawCell, Chr(10))
        ReDim nonEmpty(0 To UBound(allLines))
        lineCount = 0
        For li = 0 To UBound(allLines)
            If Trim(allLines(li)) <> "" Then
                nonEmpty(lineCount) = Trim(allLines(li))
                lineCount = lineCount + 1
            End If
        Next li
        ' lineCount = number of meaningful lines

        cleaned = ""   ' will be set by the matching rule below

        ' ----------------------------------------------------------
        ' Cleaning rules
        ' ----------------------------------------------------------
        Select Case actVal

            ' ------------------------------------------------------
            ' FUNDS TRANSFERRED
            '   Merge line1 + " " + line2, drop line3
            ' ------------------------------------------------------
            Case "Funds Transferred"
                If lineCount >= 2 Then
                    cleaned = nonEmpty(0) & " " & nonEmpty(1)
                Else
                    cleaned = nonEmpty(0)
                End If

            ' ------------------------------------------------------
            ' CHECK
            '   Keep line 1 only
            ' ------------------------------------------------------
            Case "Check"
                cleaned = nonEmpty(0)

            ' ------------------------------------------------------
            ' AUTOMATED PAYMENT
            '   2 lines -> keep line 1
            '   3 lines -> keep line 3
            ' ------------------------------------------------------
            Case "Automated Payment"
                If lineCount <= 1 Then
                    cleaned = nonEmpty(0)
                ElseIf lineCount = 2 Then
                    cleaned = nonEmpty(0)
                Else
                    cleaned = nonEmpty(2)
                End If

            ' ------------------------------------------------------
            ' FUNDS PAID
            '   1 line  -> keep as-is
            '   2+ lines -> keep line 2, strip first 5 chars
            ' ------------------------------------------------------
            Case "Funds Paid"
                If lineCount = 1 Then
                    cleaned = nonEmpty(0)
                Else
                    cleaned = nonEmpty(1)
                    If Len(cleaned) > 5 Then
                        cleaned = Mid(cleaned, 6)
                    Else
                        cleaned = ""
                    End If
                End If

            ' ------------------------------------------------------
            ' DEBIT CARD
            '   Keep line 1 only
            ' ------------------------------------------------------
            Case "Debit Card"
                cleaned = nonEmpty(0)

            ' ------------------------------------------------------
            ' WITHDRAWAL
            '   S3 (1 line)                          -> keep as-is
            '   S1 (2+ lines, line1 starts "Zelle")  -> merge line1 + " " + line2
            '   S2 (2+ lines, line1 NOT "Zelle")     -> keep line 2 only
            ' ------------------------------------------------------
            Case "Withdrawal"
                If lineCount = 1 Then
                    cleaned = nonEmpty(0)
                Else
                    firstWord = ""
                    If InStr(nonEmpty(0), " ") > 0 Then
                        firstWord = Left(nonEmpty(0), InStr(nonEmpty(0), " ") - 1)
                    Else
                        firstWord = nonEmpty(0)
                    End If
                    If LCase(firstWord) = "zelle" Then
                        cleaned = nonEmpty(0) & " " & nonEmpty(1)
                    Else
                        cleaned = nonEmpty(1)
                    End If
                End If

            ' ------------------------------------------------------
            ' DEP
            '   S1 (2 lines, line1 starts "Online") -> keep line 1
            '   S2 (1 line)                          -> keep as-is
            ' ------------------------------------------------------
            Case "DEP"
                If lineCount = 1 Then
                    cleaned = nonEmpty(0)
                Else
                    firstWord = ""
                    If InStr(nonEmpty(0), " ") > 0 Then
                        firstWord = Left(nonEmpty(0), InStr(nonEmpty(0), " ") - 1)
                    Else
                        firstWord = nonEmpty(0)
                    End If
                    If LCase(firstWord) = "online" Then
                        cleaned = nonEmpty(0)
                    Else
                        cleaned = nonEmpty(0)   ' no matching scenario - preserve line 1
                    End If
                End If

            ' ------------------------------------------------------
            ' DEPOSIT  (Institution = Bank of America)
            '   3 lines: strip first 6 chars of line2, drop line3,
            '            merge line1 + " " + trimmed line2
            '   Other institutions: leave untouched
            ' ------------------------------------------------------
            Case "DEPOSIT"
                If LCase(instVal) = LCase("Bank of America") Then
                    If lineCount >= 2 Then
                        Dim line2BofA As String
                        line2BofA = nonEmpty(1)
                        If Len(line2BofA) > 6 Then
                            line2BofA = Mid(line2BofA, 7)
                        Else
                            line2BofA = ""
                        End If
                        cleaned = nonEmpty(0) & " " & line2BofA
                    Else
                        cleaned = nonEmpty(0)
                    End If
                Else
                    GoTo NextRow   ' not BofA - leave description as-is
                End If

            ' ------------------------------------------------------
            ' DIRECT DEPOSIT  (Institution = Morgan Stanley)
            '   3 lines: drop line1, merge line2 + " " + line3
            '   Other institutions: leave untouched
            ' ------------------------------------------------------
            Case "Direct Deposit"
                If LCase(instVal) = LCase("Morgan Stanley") Then
                    If lineCount >= 3 Then
                        cleaned = nonEmpty(1) & " " & nonEmpty(2)
                    ElseIf lineCount = 2 Then
                        cleaned = nonEmpty(1)
                    Else
                        cleaned = nonEmpty(0)
                    End If
                Else
                    GoTo NextRow
                End If

            ' ------------------------------------------------------
            ' ONLINE TRANSFER  (Institution = Morgan Stanley)
            '   3 lines: keep line 2 only
            '   Other institutions: leave untouched
            ' ------------------------------------------------------
            Case "Online Transfer"
                If LCase(instVal) = LCase("Morgan Stanley") Then
                    If lineCount >= 2 Then
                        cleaned = nonEmpty(1)
                    Else
                        cleaned = nonEmpty(0)
                    End If
                Else
                    GoTo NextRow
                End If

            ' ------------------------------------------------------
            ' ZELLE PAYMENT  (Institution = Morgan Stanley)
            '   2 lines: keep line 1 only
            '   Other institutions: leave untouched
            ' ------------------------------------------------------
            Case "Zelle Payment"
                If LCase(instVal) = LCase("Morgan Stanley") Then
                    cleaned = nonEmpty(0)
                Else
                    GoTo NextRow
                End If

            ' ------------------------------------------------------
            ' FUNDS RECEIVED
            '   3 lines: keep line 3 only
            ' ------------------------------------------------------
            Case "Funds Received"
                If lineCount >= 3 Then
                    cleaned = nonEmpty(2)
                ElseIf lineCount = 2 Then
                    cleaned = nonEmpty(1)
                Else
                    cleaned = nonEmpty(0)
                End If

        End Select

        ws.Cells(r, cDescription).Value = cleaned

        NextRow:
    Next r

End Sub


' ================================================================
'  HELPER : NormaliseRowHeights
'
'  After replacing multi-line cell content with a single line,
'  the row can stay visually tall because:
'    a) WrapText is still enabled, OR
'    b) The row height was manually set by a prior format copy
'
'  Fix:
'    1. Disable WrapText on all used cells
'    2. AutoFit all rows  (same as double-clicking a row border)
'
'  This gives every row a clean, single-line height.
' ================================================================
Private Sub NormaliseRowHeights(ws As Worksheet)

    ' Turn off WrapText across the entire used range so AutoFit
    ' does not measure wrapped content and inflate row height
    ws.UsedRange.WrapText = False

    ' AutoFit every row in the used range
    ws.UsedRange.EntireRow.AutoFit

End Sub
